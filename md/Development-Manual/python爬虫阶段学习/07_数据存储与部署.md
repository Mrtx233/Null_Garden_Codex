## 1. 数据存储方案

### 1.1 如何选择？

```python
"""
数据量小（< 1万条）
  └─ JSON/CSV 文件就够了（前几个阶段的方式）

数据量大（> 1万条），需要查询、分析
  ├─ 数据结构固定 → MySQL / PostgreSQL
  ├─ 数据结构灵活 → MongoDB（爬虫数据天然适合）
  └─ 需要去重/缓存 → Redis

需要长期运行，自动化
  └─ 部署到服务器 → Scrapyd / Docker / 定时任务
"""
```

### 1.2 爬虫数据存储的最佳实践

```python
"""
推荐组合：
  1. MongoDB：存原始抓取数据（JSON 格式，不用建表）
  2. Redis：URL 去重 + 代理池 + 请求队列
  3. MySQL：最终清洗后的结构化数据（如果需要）

数据流向：
  网页 → 爬虫 → MongoDB（原始数据）→ 清洗 → MySQL（结构化）

但初学者不用这么复杂，一个 MongoDB 就够了。
"""
```

### 2.2 基础操作（pymysql）

```python
import pymysql

# ── 连接数据库 ──
conn = pymysql.connect(
    host="localhost",      # 数据库地址
    port=3306,             # 端口
    user="root",           # 用户名
    password="123456",     # 密码
    database="spider_db",  # 数据库名
    charset="utf8mb4",     # 编码
)
cursor = conn.cursor()

# ── 创建表 ──
cursor.execute("""
CREATE TABLE IF NOT EXISTS movies (
    id INT AUTO_INCREMENT PRIMARY KEY,        # 自增主键
    title VARCHAR(255) NOT NULL,              # 电影名
    score DECIMAL(3, 1),                      # 评分
    year INT,                                 # 年份
    url VARCHAR(500) UNIQUE,                  # 详情页 URL（唯一约束，防重复）
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP  # 创建时间
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
""")

# ── 插入数据 ──
sql = "INSERT INTO movies (title, score, year, url) VALUES (%s, %s, %s, %s)"
data = ("流浪地球2", 8.3, 2023, "https://example.com/movie/1")
cursor.execute(sql, data)
conn.commit()  # 提交事务
print(f"插入成功，ID: {cursor.lastrowid}")

# ── 批量插入 ──
movies = [
    ("满江红", 7.4, 2023, "https://example.com/movie/2"),
    ("哪吒", 9.0, 2025, "https://example.com/movie/3"),
    ("流浪地球2", 8.3, 2023, "https://example.com/movie/1"),  # 重复 URL
]
for movie in movies:
    try:
        cursor.execute(sql, movie)
    except pymysql.err.IntegrityError:
        print(f"重复: {movie[0]}，跳过")
conn.commit()

# ── 查询数据 ──
cursor.execute("SELECT title, score, year FROM movies WHERE score > 8.0 ORDER BY score DESC")
results = cursor.fetchall()  # 所有结果
for title, score, year in results:
    print(f"{title} | {score}分 | {year}")

# ── 更新数据 ──
cursor.execute("UPDATE movies SET score = %s WHERE title = %s", (8.5, "流浪地球2"))
conn.commit()

# ── 删除数据 ──
cursor.execute("DELETE FROM movies WHERE year < %s", (2020,))
conn.commit()

# ── 关闭连接 ──
cursor.close()
conn.close()
```

### 2.3 使用 with 上下文管理器

```python
import pymysql
from contextlib import contextmanager


@contextmanager
def get_db():
    """数据库连接上下文管理器"""
    conn = pymysql.connect(
        host="localhost",
        port=3306,
        user="root",
        password="123456",
        database="spider_db",
        charset="utf8mb4",
    )
    try:
        yield conn
        conn.commit()  # 无异常时提交
    except Exception:
        conn.rollback()  # 有异常时回滚
        raise
    finally:
        conn.close()


# 使用
with get_db() as conn:
    cursor = conn.cursor()
    cursor.execute("SELECT COUNT(*) FROM movies")
    count = cursor.fetchone()[0]
    print(f"共有 {count} 条记录")
```

### 2.4 爬虫实战：MySQL Pipeline

```python
"""
在 Scrapy 中用 Pipeline 存入 MySQL
将此代码放入 pipelines.py
"""
import pymysql


class MySQLPipeline:
    """Scrapy Pipeline：数据存入 MySQL"""

    def open_spider(self, spider):
        """爬虫启动时连接数据库"""
        self.conn = pymysql.connect(
            host=spider.settings.get("MYSQL_HOST", "localhost"),
            port=spider.settings.get("MYSQL_PORT", 3306),
            user=spider.settings.get("MYSQL_USER", "root"),
            password=spider.settings.get("MYSQL_PASSWORD", ""),
            database=spider.settings.get("MYSQL_DATABASE", "spider"),
            charset="utf8mb4",
        )
        self.cursor = self.conn.cursor()

    def close_spider(self, spider):
        """爬虫关闭时关闭连接"""
        self.conn.close()

    def process_item(self, item, spider):
        """处理每条数据"""
        sql = """
            INSERT INTO movies (title, score, year, url)
            VALUES (%s, %s, %s, %s)
            ON DUPLICATE KEY UPDATE score=VALUES(score)
        """
        # ON DUPLICATE KEY UPDATE: 如果 URL 重复，更新评分
        data = (item["title"], item["score"], item["year"], item["url"])
        self.cursor.execute(sql, data)
        self.conn.commit()
        return item
```

### 2.5 使用 SQLAlchemy（ORM，推荐）

```python
"""
ORM = Object Relational Mapping
让你用 Python 对象操作数据库，不用写 SQL
"""

# pip install sqlalchemy pymysql

from sqlalchemy import create_engine, Column, Integer, String, Float, DateTime, func
from sqlalchemy.orm import declarative_base, sessionmaker

# ── 定义模型 ──
Base = declarative_base()


class Movie(Base):
    __tablename__ = "movies"

    id = Column(Integer, primary_key=True, autoincrement=True)
    title = Column(String(255), nullable=False)
    score = Column(Float)
    year = Column(Integer)
    url = Column(String(500), unique=True)  # 唯一约束
    created_at = Column(DateTime, default=func.now())

    def __repr__(self):
        return f"<Movie {self.title}>"


# ── 连接数据库 ──
engine = create_engine(
    "mysql+pymysql://root:123456@localhost:3306/spider_db?charset=utf8mb4",
    echo=False,  # True 会打印 SQL 语句
)
# 创建所有表
Base.metadata.create_all(engine)

# 创建会话
Session = sessionmaker(bind=engine)
session = Session()


# ── 增 ──
movie = Movie(title="流浪地球2", score=8.3, year=2023, url="https://example.com/1")
session.add(movie)
session.commit()

# 批量增
session.add_all([
    Movie(title="满江红", score=7.4, year=2023, url="https://example.com/2"),
    Movie(title="哪吒", score=9.0, year=2025, url="https://example.com/3"),
])
session.commit()


# ── 查 ──
# 查所有
movies = session.query(Movie).all()

# 条件查询
good_movies = session.query(Movie).filter(Movie.score > 8.0).order_by(Movie.score.desc()).all()
for m in good_movies:
    print(m.title, m.score)

# 查一条
movie = session.query(Movie).filter_by(title="流浪地球2").first()

# 统计
count = session.query(Movie).count()


# ── 改 ──
movie = session.query(Movie).filter_by(title="流浪地球2").first()
if movie:
    movie.score = 8.5
    session.commit()


# ── 删 ──
session.query(Movie).filter(Movie.year < 2020).delete()
session.commit()


# ── 关闭 ──
session.close()
```

## 3. 非关系型数据库 — MongoDB

### 3.1 为什么爬虫适合 MongoDB？

```python
"""
MongoDB 的优点：
1. 存 JSON 原生支持（爬虫数据就是 JSON，不用建表）
2. 灵活 schema（每条数据的字段可以不一样）
3. 写入速度快（爬虫场景非常适合）
4. 横向扩展容易

说人话：爬虫抓到的数据直接往里扔，不用设计表结构。
"""
```

### 3.3 基础操作

```python
from pymongo import MongoClient

# ── 连接 ──
client = MongoClient("mongodb://localhost:27017")
# 如果有密码：MongoClient("mongodb://user:password@localhost:27017")

# 选择/创建数据库
db = client["spider_db"]

# 选择/创建集合（类似 MySQL 的表）
collection = db["movies"]


# ── 插入 ──
# insert_one：插入一条
movie = {
    "title": "流浪地球2",
    "score": 8.3,
    "year": 2023,
    "url": "https://example.com/movie/1",
    "director": "郭帆",
    "actors": ["吴京", "刘德华"],
}
result = collection.insert_one(movie)
print(f"插入 ID: {result.inserted_id}")

# insert_many：批量插入
movies = [
    {"title": "满江红", "score": 7.4, "year": 2023, "url": "https://example.com/2"},
    {"title": "哪吒", "score": 9.0, "year": 2025, "url": "https://example.com/3"},
]
result = collection.insert_many(movies)
print(f"插入了 {len(result.inserted_ids)} 条")


# ── 查询 ──
# 查所有
for movie in collection.find():
    print(movie["title"], movie["score"])

# 条件查询
for movie in collection.find({"score": {"$gt": 8.0}}):
    print(movie["title"])

# 排序、限制
for movie in collection.find().sort("score", -1).limit(10):
    print(movie["title"], movie["score"])

# 查一条
movie = collection.find_one({"title": "流浪地球2"})

# 只返回特定字段
for movie in collection.find({}, {"title": 1, "score": 1}):
    print(movie)


# ── 更新 ──
collection.update_one(
    {"title": "流浪地球2"},
    {"$set": {"score": 8.5}}  # $set 只修改指定字段
)

collection.update_many(
    {"year": 2023},
    {"$set": {"tag": "2023年电影"}}
)


# ── 删除 ──
collection.delete_one({"title": "满江红"})
collection.delete_many({"year": {"$lt": 2020}})


# ── 去重计数 ──
count = collection.count_documents({"score": {"$gt": 8.0}})
print(f"高分电影: {count} 部")


# ── 创建索引（提高查询速度）──
collection.create_index("title")            # 单字段索引
collection.create_index([("score", -1)])     # 降序索引
collection.create_index("url", unique=True)  # 唯一索引（防重复）
```

### 3.4 爬虫实战：MongoDB Pipeline

```python
"""
Scrapy Pipeline 存入 MongoDB
"""
import pymongo


class MongoPipeline:
    """Scrapy Pipeline：数据存入 MongoDB"""

    def open_spider(self, spider):
        self.client = MongoClient(spider.settings.get("MONGO_URI", "mongodb://localhost:27017"))
        self.db = self.client[spider.settings.get("MONGO_DATABASE", "spider")]
        self.collection = self.db[spider.settings.get("MONGO_COLLECTION", "items")]
        # 创建唯一索引防重复（假设 url 是去重依据）
        self.collection.create_index("url", unique=True)

    def close_spider(self, spider):
        self.client.close()

    def process_item(self, item, spider):
        try:
            self.collection.insert_one(dict(item))
        except pymongo.errors.DuplicateKeyError:
            # URL 重复，更新数据
            self.collection.update_one(
                {"url": item["url"]},
                {"$set": dict(item)}
            )
        return item
```

### 3.5 不用 Scrapy 时的 MongoDB 存储

```python
"""
如果不使用 Scrapy，直接存 MongoDB 也很简单
"""
from pymongo import MongoClient, errors


class MongoStorage:
    """独立的 MongoDB 存储类"""

    def __init__(self, uri="mongodb://localhost:27017", db_name="spider", collection_name="data"):
        self.client = MongoClient(uri)
        self.db = self.client[db_name]
        self.collection = self.db[collection_name]
        # URL 唯一索引
        self.collection.create_index("url", unique=True)

    def save(self, item):
        """保存单条数据"""
        try:
            self.collection.insert_one(item)
            return True
        except errors.DuplicateKeyError:
            # 重复则更新
            self.collection.update_one(
                {"url": item["url"]},
                {"$set": item}
            )
            return False

    def save_many(self, items):
        """批量保存"""
        for item in items:
            self.save(item)

    def close(self):
        self.client.close()


# 使用
storage = MongoStorage()
storage.save({"title": "流浪地球2", "score": 8.3, "url": "https://example.com/1"})
storage.close()
```

## 4. 缓存数据库 — Redis

### 4.1 Redis 在爬虫中的作用

```python
"""
Redis 是内存数据库，速度极快。在爬虫中主要用于：

1. URL 去重：用 Set 存已爬过的 URL（上千万个也不怕）
2. 请求队列：用 List 做 FIFO 队列（Scrapy-Redis 的核心）
3. 代理池：用 Set 存可用代理
4. 计数器：用 INCR 命令（统计爬取数量）
5. 缓存：存已抓取的页面内容，避免重复请求
"""
```

### 4.2 安装与连接

```bash
pip install redis
```

### 4.3 基础操作

```python
import redis

# ── 连接 ──
r = redis.Redis(
    host="localhost",
    port=6379,
    password="",     # 没有密码就留空
    db=0,            # 数据库编号（0-15）
    decode_responses=True,  # 自动解码为字符串（不加返回字节）
)

# ── 测试连接 ──
print(r.ping())  # True


# ── String（字符串）──
r.set("key", "value")              # 设置
r.set("count", 0)
print(r.get("key"))                # "value"
r.incr("count")                    # 递增 +1 → 1
r.incrby("count", 5)               # 递增 +5 → 6
print(r.get("count"))              # "6"
r.expire("key", 3600)              # 设置过期时间（秒）


# ── List（列表，用作队列）──
r.lpush("url_queue", "https://a.com")  # 左边推入
r.rpush("url_queue", "https://b.com")  # 右边推入
r.lpush("url_queue", "https://c.com", "https://d.com")  # 批量

length = r.llen("url_queue")            # 队列长度
url = r.lpop("url_queue")               # 左边弹出（FIFO）
url = r.brpop("url_queue", timeout=5)   # 阻塞弹出（等 5 秒）
# brpop 用于多消费者场景：没有任务就等，有任务立马取


# ── Set（集合，用于去重）──
r.sadd("visited_urls", "https://a.com")           # 添加
r.sadd("visited_urls", "https://a.com")           # 重复添加不会生效
r.sadd("visited_urls", "https://b.com", "https://c.com")  # 批量

is_visited = r.sismember("visited_urls", "https://a.com")  # True
all_visited = r.smembers("visited_urls")           # 所有已访问的 URL
count = r.scard("visited_urls")                    # 数量


# ── 爬虫实战：URL 去重 ──
class URLFilter:
    """基于 Redis 的 URL 去重器"""

    def __init__(self, redis_client, key="url_filter"):
        self.redis = redis_client
        self.key = key

    def is_duplicate(self, url):
        """检查 URL 是否已访问，未访问则标记并返回 False"""
        return self.redis.sadd(self.key, url) == 0
        # sadd 返回 1=新添加（未访问），0=已存在（重复）

    def clear(self):
        self.redis.delete(self.key)


# 使用
rf = URLFilter(r)
urls = ["https://a.com", "https://b.com", "https://a.com", "https://c.com"]

for url in urls:
    if rf.is_duplicate(url):
        print(f"跳过（重复）: {url}")
    else:
        print(f"抓取: {url}")
# 输出：
# 抓取: https://a.com
# 抓取: https://b.com
# 跳过（重复）: https://a.com
# 抓取: https://c.com


# ── 爬虫实战：分布式代理池 ──
class ProxyPool:
    """基于 Redis 的代理池"""

    def __init__(self, redis_client, key="proxy_pool"):
        self.redis = redis_client
        self.key = key

    def add(self, proxy):
        """添加代理"""
        self.redis.sadd(self.key, proxy)

    def get_random(self):
        """随机获取一个代理"""
        return self.redis.srandmember(self.key)

    def remove(self, proxy):
        """移除失效代理"""
        self.redis.srem(self.key, proxy)

    def count(self):
        """可用代理数量"""
        return self.redis.scard(self.key)
```

"""
如果不想用 crontab（比如在 Windows 上），用 APScheduler。
"""

# pip install apscheduler

from apscheduler.schedulers.blocking import BlockingScheduler
import requests
import json
from datetime import datetime


def crawl_job():
    """定时执行的爬虫任务"""
    print(f"[{datetime.now()}] 开始抓取...")
    try:
        response = requests.get("https://example.com/data", timeout=10)
        data = response.json()
        with open(f"data_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json", "w") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        print(f"抓取完成，{len(data)} 条数据")
    except Exception as e:
        print(f"抓取失败: {e}")


# 创建调度器
scheduler = BlockingScheduler()

# 每天凌晨 2 点运行
scheduler.add_job(crawl_job, "cron", hour=2, minute=0)

# 每 6 小时运行一次
# scheduler.add_job(crawl_job, "interval", hours=6)

# 每周一早上 9 点
# scheduler.add_job(crawl_job, "cron", day_of_week="mon", hour=9, minute=0)

print("定时任务已启动...")
scheduler.start()

### 5.6 完整部署脚本示例

```python
"""
一个完整的上线方案：爬虫 + MongoDB + 定时任务
"""
import json
import logging
from datetime import datetime
from pymongo import MongoClient, errors

# 配置日志
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[
        logging.FileHandler("crawler.log", encoding="utf-8"),
        logging.StreamHandler(),
    ]
)
logger = logging.getLogger(__name__)


class ProductionCrawler:
    """生产环境爬虫"""

    def __init__(self):
        # MongoDB 连接
        self.mongo_uri = "mongodb://localhost:27017"
        self.db_name = "spider_prod"
        self.collection_name = "data"
        self.client = None
        self.collection = None

    def connect_db(self):
        """连接数据库"""
        self.client = MongoClient(self.mongo_uri)
        self.collection = self.client[self.db_name][self.collection_name]
        self.collection.create_index("url", unique=True)
        logger.info("MongoDB 连接成功")

    def fetch_data(self):
        """抓取数据"""
        logger.info("开始抓取数据...")
        # 实际爬虫逻辑
        results = []
        for i in range(5):
            results.append({
                "title": f"数据 {i}",
                "value": i * 10,
                "url": f"https://example.com/{i}",
                "crawled_at": datetime.now().isoformat(),
            })
        return results

    def save_data(self, items):
        """保存数据到 MongoDB"""
        saved = 0
        updated = 0
        for item in items:
            try:
                self.collection.insert_one(item)
                saved += 1
            except errors.DuplicateKeyError:
                self.collection.update_one(
                    {"url": item["url"]},
                    {"$set": item}
                )
                updated += 1

        logger.info(f"新增 {saved} 条，更新 {updated} 条")

    def run(self):
        """运行爬虫"""
        try:
            self.connect_db()
            data = self.fetch_data()
            self.save_data(data)
            logger.info("本次运行完成")
        except Exception as e:
            logger.error(f"运行失败: {e}", exc_info=True)
        finally:
            if self.client:
                self.client.close()


# 直接运行
if __name__ == "__main__":
    crawler = ProductionCrawler()
    crawler.run()

# 如果配合 APScheduler 定时运行
# scheduler = BlockingScheduler()
# scheduler.add_job(lambda: ProductionCrawler().run(), "interval", hours=6)
# scheduler.start()
```

## 学习建议

1. **从文件存储开始**：JSON/CSV 在开发阶段完全够用，不要一上来就上数据库。

2. **MongoDB 是爬虫的最佳拍档**：数据格式天然匹配，不用建表，写入快。强烈推荐作为爬虫的数据存储首选。

3. **Redis 用于去重和队列**：Redis 不是必选项，但当你要爬百万级 URL 时，它的去重能力是文件方式无法比拟的。

4. **Docker 是现代部署的标准**：虽然 Docker 本身有一定学习成本，但它解决了"在我机器上能跑"的问题。

5. **定时任务注意日志**：部署到服务器后，你无法实时看到输出。一定要加日志（logging），出了问题才能查。

6. **监控与报警**：生产环境爬虫要考虑：
   - 爬虫挂了怎么办？（自动重启）
   - 数据量异常怎么办？（监控告警）
   - IP 被封了怎么办？（代理自动切换）

import { onMounted, reactive, ref } from "vue"

import {
  getMyProfileApi,

const loading = ref(false)
const saving = ref(false)
const message = ref("")

const profile = ref({})

const form = reactive({
  level: "",
  gender: "",
  birthday: "",
  height_cm: "",
  weight_kg: "",
  avatar_url: "",
  intro: "",
  join_time: "",
  expire_date: ""

function datetimeToInputValue(value) {
  if (!value) {
    return ""
  }

  return value.replace(" ", "T").slice(0, 16)
}

function inputValueToBackendDatetime(value) {
  if (!value) {
    return null
  }

  return value.replace("T", " ") + ":00"
}

function fillForm(data) {
  profile.value = data

  form.level = data.level ?? ""
  form.gender = data.gender ?? ""
  form.birthday = data.birthday ?? ""
  form.height_cm = data.height_cm ?? ""
  form.weight_kg = data.weight_kg ?? ""
  form.avatar_url = data.avatar_url ?? ""
  form.intro = data.intro ?? ""
  form.join_time = datetimeToInputValue(data.join_time)

}

async function loadProfile() {
  try {
    loading.value = true
    message.value = ""

    const res = await getMyProfileApi()

    if (res.code === 200) {
      fillForm(res.data)
    } else {
      message.value = res.message || "获取档案失败"
    }
  } catch (error) {
    if (error.response && error.response.data) {
      message.value = error.response.data.message || "获取档案失败"
    } else {
      message.value = "获取档案失败"
    }
  } finally {
    loading.value = false
  }
}

async function handleSave() {
  try {
    saving.value = true
    message.value = ""

    const payload = {
      level: form.level === "" ? null : Number(form.level),
      gender: form.gender === "" ? null : Number(form.gender),
      birthday: form.birthday || null,
      height_cm: form.height_cm === "" ? null : String(form.height_cm),
      weight_kg: form.weight_kg === "" ? null : String(form.weight_kg),
      avatar_url: form.avatar_url || "",
      intro: form.intro || "",
      join_time: form.join_time
        ? inputValueToBackendDatetime(form.join_time)
        : null,
      expire_date: form.expire_date || null
    }

    const res = await updateMyProfileApi(payload)

    if (res.code === 200) {
      fillForm(res.data)
      message.value = "保存成功"
    } else {
      message.value = res.message || "保存失败"
    }
  } catch (error) {
    if (error.response && error.response.data) {
      message.value = error.response.data.message || "保存失败"
    } else {
      message.value = "保存失败"
    }
  } finally {
    saving.value = false
  }
}

# 三、修改阶段 17 的路由文件

所以现在只加 `Profile`。

完整修改后应该是：

```javascript
import { createRouter, createWebHistory } from "vue-router"

import { useAuthStore } from "../stores/auth"

import Login from "../views/Login.vue"
import Home from "../views/Home.vue"
import UserList from "../views/UserList.vue"
import StoreList from "../views/StoreList.vue"
import CourseList from "../views/CourseList.vue"
import ExerciseList from "../views/ExerciseList.vue"
import MyFavorites from "../views/MyFavorites.vue"
import Profile from "../views/Profile.vue"
import Forbidden from "../views/Forbidden.vue"

const routes = [
  {
    path: "/login",
    component: Login
  },
  {
    path: "/",
    component: Home,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/users",
    component: UserList,
    meta: {
      requiresAuth: true,
      permission: "user:list"
    }
  },
  {
    path: "/store",
    component: StoreList,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/course",
    component: CourseList,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/exercise",
    component: ExerciseList,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/favorites",
    component: MyFavorites,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/profile",
    component: Profile,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/403",
    component: Forbidden
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

router.beforeEach((to, from, next) => {
  const authStore = useAuthStore()

  if (to.meta.requiresAuth && !authStore.isLogin) {
    next("/login")
    return
  }

  if (to.path === "/login" && authStore.isLogin) {
    next("/")
    return
  }

  if (to.meta.permission && !authStore.hasPermission(to.meta.permission)) {
    next("/403")
    return
  }

  next()
})

export default router
```

# 五、为什么“我的档案”不加权限判断？
这里不要写：

```vue
v-if="authStore.hasPermission('profile:me')"
```

因为当前设计中：

```latex
/profile 是当前登录用户自己的档案
```

它只需要登录即可访问。

所以路由中写这个就够了：

```javascript
meta: {
  requiresAuth: true
}
```

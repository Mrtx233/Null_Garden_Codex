本阶段目标：

```latex
1. 查询省份/区域列表
2. 查询门店列表
3. 查询门店详情
4. 查询当前登录用户关联的门店
```

本阶段基于你前面设计的这三张表：

```latex
t_store_province   门店所属省份/区域表
t_store            门店基础信息表
y_user_store       用户与门店关联表
```

这三张表来自你上传的业务表结构，其中 `t_store` 通过 `province_id` 关联省份表，`y_user_store` 通过 `user_id`、`role_id`、`store_id` 关联用户、角色和门店。

```latex
新增门店

删除门店
Vue 页面

```

## 2. 门店列表
请求：

```latex
GET /api/store/stores/
```

支持筛选：

```latex
province_id     按省份筛选
city            按城市筛选
store_type      按门店类型筛选
keyword         按门店名称模糊搜索

```

例如：

```latex
GET /api/store/stores/?province_id=1
GET /api/store/stores/?keyword=广州
GET /api/store/stores/?store_type=1
```

---

## 3. 门店详情
请求：

```latex
GET /api/store/stores/1/
```

返回单个门店的详细信息。

---

## 4. 当前用户关联门店
请求：

```latex
GET /api/store/my-stores/
```

后端会根据 JWT token 找到当前用户，然后查询：

```latex
y_user_store
```

返回这个用户关联的门店。

例如：

```latex
admin -> 广州天河铁馆
zhangsan -> 上海静安私教馆
```

# 三、修改 `store/views.py`
打开文件：

```latex
backend/store/views.py
```

把内容替换成下面完整代码。

## `backend/store/views.py`
```python
from django.views import View

from rbac.auth import login_required
from rbac.responses import json_response

from .models import StoreProvince, Store, UserStore


def format_decimal_value(value):
    """
    Decimal 转字符串，避免 JSON 序列化问题。
    """

    if value is None:
        return None

    return str(value)


def format_datetime_value(value):
    """
    datetime 转字符串。
    """

    if value is None:
        return None

    return value.strftime("%Y-%m-%d %H:%M:%S")


def serialize_province(province):
    """
    省份对象转 JSON 字典。
    """

    return {
        "province_id": province.province_id,
        "province_name": province.province_name,
        "center_lng": format_decimal_value(province.center_lng),
        "center_lat": format_decimal_value(province.center_lat),
    }


def serialize_store(store):
    """
    门店对象转 JSON 字典。
    """

    return {
        "store_id": store.store_id,
        "store_name": store.store_name,
        "store_type": store.store_type,
        "store_type_text": store.get_store_type_display(),

        "province_id": store.province_id,
        "province_name": store.province_name or store.province.province_name,

        "city": store.city,
        "district": store.district,
        "address": store.address,

        "store_phone": store.store_phone,
        "store_image_url": store.store_image_url,
        "store_introduction": store.store_introduction,
        "business_hours": store.business_hours,

        "is_operating": store.is_operating,
        "is_operating_text": store.get_is_operating_display(),

        "store_lng": format_decimal_value(store.store_lng),
        "store_lat": format_decimal_value(store.store_lat),
    }


def serialize_user_store(user_store):
    """
    用户门店关系对象转 JSON 字典。
    """

    store = user_store.store
    user = user_store.user
    role = user_store.role

    return {
        "id": user_store.id,

        "user_id": user.user_id,
        "username": user.username,
        "real_name": user.real_name,

        "role_id": role.id,
        "role_code": role.role_code,
        "role_name": role.role_name,

        "store": serialize_store(store),

        "created_at": format_datetime_value(user_store.created_at),
    }


class StoreProvinceListView(View):
    """
    省份/区域列表接口

    GET /api/store/provinces/
    """

    @login_required
    def get(self, request):
        provinces = StoreProvince.objects.all().order_by("province_id")

        data = []

        for province in provinces:
            data.append(serialize_province(province))

        return json_response(
            data=data,
            message="获取省份列表成功",
        )


class StoreListView(View):
    """
    门店列表接口

    GET /api/store/stores/

    可选查询参数：
    province_id
    city
    store_type
    keyword
    is_operating
    """

    @login_required
    def get(self, request):
        queryset = Store.objects.select_related("province").all().order_by("store_id")

        province_id = request.GET.get("province_id")
        city = request.GET.get("city")
        store_type = request.GET.get("store_type")
        keyword = request.GET.get("keyword")
        is_operating = request.GET.get("is_operating")

        if province_id:
            queryset = queryset.filter(province_id=province_id)

        if city:
            queryset = queryset.filter(city=city)

        if store_type:
            queryset = queryset.filter(store_type=store_type)

        if keyword:
            queryset = queryset.filter(store_name__icontains=keyword)

        if is_operating in ["0", "1"]:
            queryset = queryset.filter(is_operating=int(is_operating))

        data = []

        for store in queryset:
            data.append(serialize_store(store))

        return json_response(
            data=data,
            message="获取门店列表成功",
        )


class StoreDetailView(View):
    """
    门店详情接口

    GET /api/store/stores/<store_id>/
    """

    @login_required
    def get(self, request, store_id):
        store = Store.objects.select_related("province").filter(
            store_id=store_id
        ).first()

        if store is None:
            return json_response(
                message="门店不存在",
                code=404,
                status=404,
            )

        return json_response(
            data=serialize_store(store),
            message="获取门店详情成功",
        )


class MyStoreListView(View):
    """
    当前登录用户关联门店接口

    GET /api/store/my-stores/
    """

    @login_required
    def get(self, request):
        current_user = request.current_user

        user_stores = UserStore.objects.select_related(
            "user",
            "role",
            "store",
            "store__province",
        ).filter(
            user=current_user
        ).order_by("-created_at")

        data = []

        for item in user_stores:
            data.append(serialize_user_store(item))

        return json_response(
            data=data,
            message="获取当前用户关联门店成功",
        )
```

本阶段目标：

```latex
1. Vue3 调用门店接口
2. Vue3 调用课程接口
3. Vue3 调用动作接口
4. Vue3 展示我的课程收藏
5. Vue3 展示我的动作收藏
6. Vue3 实现收藏 / 取消收藏
```

这些页面对应你后端已经建立的业务模块：`store` 管理省份、门店、用户门店关系；`course` 管理课程分类、课程、课程收藏、门店课程排期；`exercise` 管理动作分类、动作库、动作收藏。你上传的表结构里也包含了这些业务表。

本阶段不做：

```latex
分页
复杂搜索
新增 / 编辑 / 删除门店
新增 / 编辑 / 删除课程
新增 / 编辑 / 删除动作
Element Plus UI
```

先用最简单的原生 Vue 页面跑通接口。

const myStores = ref([])

  province_id: "",
  store_type: "",
  is_operating: "",
  keyword: ""
})

async function loadProvinces() {
  const res = await getProvinceListApi()

  if (res.code === 200) {
    provinces.value = res.data
  } else {
    alert(res.message || "获取省份失败")
  }
}

async function loadStores() {
  const params = {}

  if (query.province_id) {
    params.province_id = query.province_id
  }

  if (query.store_type) {
    params.store_type = query.store_type
  }

  if (query.is_operating !== "") {
    params.is_operating = query.is_operating
  }

  if (query.keyword) {
    params.keyword = query.keyword
  }

  const res = await getStoreListApi(params)

  if (res.code === 200) {
    stores.value = res.data
  } else {
    alert(res.message || "获取门店失败")
  }
}

async function loadMyStores() {
  const res = await getMyStoreListApi()

  if (res.code === 200) {
    myStores.value = res.data
  } else {
    alert(res.message || "获取我的门店失败")
  }
}

function resetQuery() {
  query.province_id = ""
  query.store_type = ""
  query.is_operating = ""
  query.keyword = ""

  loadStores()
}

onMounted(async () => {
  await loadProvinces()
  await loadStores()
  await loadMyStores()
})

# 十、修改路由文件
打开：

```latex
frontend/src/router/index.js
```

替换成下面完整代码：

```javascript
import { createRouter, createWebHistory } from "vue-router"

import { useAuthStore } from "../stores/auth"

import Login from "../views/Login.vue"
import Home from "../views/Home.vue"
import UserList from "../views/UserList.vue"
import StoreList from "../views/StoreList.vue"
import CourseList from "../views/CourseList.vue"
import ExerciseList from "../views/ExerciseList.vue"
import MyFavorites from "../views/MyFavorites.vue"
import Forbidden from "../views/Forbidden.vue"

const routes = [
  {
    path: "/login",
    component: Login
  },
  {
    path: "/",
    component: Home,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/users",
    component: UserList,
    meta: {
      requiresAuth: true,
      permission: "user:list"
    }
  },
  {
    path: "/store",
    component: StoreList,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/course",
    component: CourseList,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/exercise",
    component: ExerciseList,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/favorites",
    component: MyFavorites,
    meta: {
      requiresAuth: true
    }
  },
  {
    path: "/403",
    component: Forbidden
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

router.beforeEach((to, from, next) => {
  const authStore = useAuthStore()

  if (to.meta.requiresAuth && !authStore.isLogin) {
    next("/login")
    return
  }

  if (to.path === "/login" && authStore.isLogin) {
    next("/")
    return
  }

  if (to.meta.permission && !authStore.hasPermission(to.meta.permission)) {
    next("/403")
    return
  }

  next()
})

export default router
```

## 2. 页面请求接口返回 401
说明 token 没有带上。

```latex
frontend/src/utils/request.js
```

里面必须有：

```javascript
config.headers.Authorization = `Bearer ${accessToken}`
```

同时检查浏览器 Local Storage 里有没有：

```latex
access_token
```

python manage.py init_business_data

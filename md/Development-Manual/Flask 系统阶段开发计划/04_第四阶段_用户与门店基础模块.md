## 阶段导语

本阶段基于第三阶段 RBAC（Role-Based Access Control，基于角色的访问控制）权限模块继续开发，实现用户管理、用户档案管理、省份/区域管理、门店管理与用户绑定门店。完成本阶段后，后台可以新增、编辑、禁用用户，可以维护用户扩展档案与省份/区域、门店数据；用户与门店之间为多对多关系（Many-to-Many），用户可以绑定一个或多个门店；所有页面均受 `user:manage` 或 `store:manage` 权限控制。

## 3. 用户管理路由

文件：`app/blueprints/admin/user_routes.py`

```python
from datetime import datetime

from flask import flash, redirect, render_template, request, url_for

from app.extensions import db
from app.models import Role, Store, User, UserProfile, UserStore
from app.utils.auth import permission_required
from . import admin_bp


@admin_bp.route("/users")
@permission_required("user:manage")
def users():
    keyword = request.args.get("keyword", "").strip()
    query = User.query
    if keyword:
        query = query.filter(User.username.like(f"%{keyword}%") | User.real_name.like(f"%{keyword}%"))
    items = query.order_by(User.user_id.desc()).all()
    return render_template("admin/users.html", users=items, keyword=keyword)


@admin_bp.route("/users/create", methods=["GET", "POST"])
@permission_required("user:manage")
def user_create():
    roles = Role.query.order_by(Role.id.asc()).all()
    if request.method == "POST":
        user = User(
            role_id=int(request.form["role_id"]),
            username=request.form.get("username", "").strip(),
            real_name=request.form.get("real_name", "").strip(),
            phone=request.form.get("phone", "").strip(),
            status=int(request.form.get("status", 1)),
        )
        user.set_password(request.form.get("password", "123456"))
        db.session.add(user)
        db.session.flush()
        db.session.add(UserProfile(user_id=user.user_id, create_time=datetime.utcnow()))
        db.session.commit()
        flash("用户已创建。", "success")
        return redirect(url_for("admin.users"))
    return render_template("admin/user_form.html", user=None, roles=roles)


@admin_bp.route("/users/<int:user_id>/edit", methods=["GET", "POST"])
@permission_required("user:manage")
def user_edit(user_id):
    user = User.query.get_or_404(user_id)
    roles = Role.query.order_by(Role.id.asc()).all()
    if request.method == "POST":
        user.role_id = int(request.form["role_id"])
        user.real_name = request.form.get("real_name", "").strip()
        user.phone = request.form.get("phone", "").strip()
        user.status = int(request.form.get("status", 1))
        password = request.form.get("password", "").strip()
        if password:
            user.set_password(password)

        profile = user.profile or UserProfile(user_id=user.user_id)
        profile.level = request.form.get("level") or None
        profile.gender = request.form.get("gender") or None
        profile.birthday = request.form.get("birthday") or None
        profile.height_cm = request.form.get("height_cm") or None
        profile.weight_kg = request.form.get("weight_kg") or None
        profile.avatar_url = request.form.get("avatar_url", "").strip()
        profile.intro = request.form.get("intro", "").strip()
        profile.join_time = request.form.get("join_time") or None
        profile.expire_date = request.form.get("expire_date") or None
        db.session.add(profile)
        db.session.commit()
        flash("用户已更新。", "success")
        return redirect(url_for("admin.users"))
    return render_template("admin/user_form.html", user=user, roles=roles)


@admin_bp.route("/users/<int:user_id>/stores", methods=["GET", "POST"])
@permission_required("user:manage")
def user_stores(user_id):
    user = User.query.get_or_404(user_id)
    stores = Store.query.order_by(Store.store_id.asc()).all()
    if request.method == "POST":
        store_ids = {int(item) for item in request.form.getlist("store_ids")}
        UserStore.query.filter_by(user_id=user.user_id).delete()
        for store_id in store_ids:
            db.session.add(UserStore(user_id=user.user_id, role_id=user.role_id, store_id=store_id))
        db.session.commit()
        flash("用户门店绑定已保存。", "success")
        return redirect(url_for("admin.users"))

    selected_ids = {item.store_id for item in UserStore.query.filter_by(user_id=user.user_id).all()}
    return render_template("admin/user_stores.html", user=user, stores=stores, selected_ids=selected_ids)
```

- **要点**：用户列表通过 `username` 与 `real_name` 的模糊匹配实现关键词搜索，结果按 `user_id` 倒序排列。
- **要点**：新增用户时同步创建 UserProfile 记录，保证用户与档案之间的一对一关系（One-to-One）成立。
- **要点**：编辑用户时密码留空则不修改，避免误覆盖已有密码。
- **要点**：绑定门店采用"先删除全部旧关联、再按勾选结果重建"的覆盖式保存策略，逻辑简单且结果确定。

## 4. 省份和门店管理路由

文件：`app/blueprints/admin/store_routes.py`

```python
from flask import flash, redirect, render_template, request, url_for

from app.extensions import db
from app.models import Store, StoreProvince
from app.utils.auth import permission_required
from . import admin_bp


@admin_bp.route("/provinces")
@permission_required("store:manage")
def provinces():
    items = StoreProvince.query.order_by(StoreProvince.province_id.desc()).all()
    return render_template("admin/provinces.html", provinces=items)


@admin_bp.route("/provinces/create", methods=["GET", "POST"])
@permission_required("store:manage")
def province_create():
    if request.method == "POST":
        province = StoreProvince(
            province_name=request.form.get("province_name", "").strip(),
            center_lng=request.form.get("center_lng") or None,
            center_lat=request.form.get("center_lat") or None,
        )
        db.session.add(province)
        db.session.commit()
        flash("省份/区域已创建。", "success")
        return redirect(url_for("admin.provinces"))
    return render_template("admin/province_form.html", province=None)


@admin_bp.route("/provinces/<int:province_id>/edit", methods=["GET", "POST"])
@permission_required("store:manage")
def province_edit(province_id):
    province = StoreProvince.query.get_or_404(province_id)
    if request.method == "POST":
        province.province_name = request.form.get("province_name", "").strip()
        province.center_lng = request.form.get("center_lng") or None
        province.center_lat = request.form.get("center_lat") or None
        db.session.commit()
        flash("省份/区域已更新。", "success")
        return redirect(url_for("admin.provinces"))
    return render_template("admin/province_form.html", province=province)


@admin_bp.route("/stores")
@permission_required("store:manage")
def stores():
    items = Store.query.order_by(Store.store_id.desc()).all()
    return render_template("admin/stores.html", stores=items)


@admin_bp.route("/stores/create", methods=["GET", "POST"])
@permission_required("store:manage")
def store_create():
    provinces = StoreProvince.query.order_by(StoreProvince.province_id.asc()).all()
    if request.method == "POST":
        province = StoreProvince.query.get_or_404(int(request.form["province_id"]))
        store = Store(
            store_name=request.form.get("store_name", "").strip(),
            store_type=int(request.form.get("store_type", 1)),
            province_id=province.province_id,
            province_name=province.province_name,
            city=request.form.get("city", "").strip(),
            district=request.form.get("district", "").strip(),
            address=request.form.get("address", "").strip(),
            store_phone=request.form.get("store_phone", "").strip(),
            store_image_url=request.form.get("store_image_url", "").strip(),
            store_introduction=request.form.get("store_introduction", "").strip(),
            business_hours=request.form.get("business_hours", "").strip(),
            is_operating=int(request.form.get("is_operating", 1)),
            store_lng=request.form.get("store_lng") or None,
            store_lat=request.form.get("store_lat") or None,
        )
        db.session.add(store)
        db.session.commit()
        flash("门店已创建。", "success")
        return redirect(url_for("admin.stores"))
    return render_template("admin/store_form.html", store=None, provinces=provinces)


@admin_bp.route("/stores/<int:store_id>/edit", methods=["GET", "POST"])
@permission_required("store:manage")
def store_edit(store_id):
    store = Store.query.get_or_404(store_id)
    provinces = StoreProvince.query.order_by(StoreProvince.province_id.asc()).all()
    if request.method == "POST":
        province = StoreProvince.query.get_or_404(int(request.form["province_id"]))
        store.store_name = request.form.get("store_name", "").strip()
        store.store_type = int(request.form.get("store_type", 1))
        store.province_id = province.province_id
        store.province_name = province.province_name
        store.city = request.form.get("city", "").strip()
        store.district = request.form.get("district", "").strip()
        store.address = request.form.get("address", "").strip()
        store.store_phone = request.form.get("store_phone", "").strip()
        store.store_image_url = request.form.get("store_image_url", "").strip()
        store.store_introduction = request.form.get("store_introduction", "").strip()
        store.business_hours = request.form.get("business_hours", "").strip()
        store.is_operating = int(request.form.get("is_operating", 1))
        store.store_lng = request.form.get("store_lng") or None
        store.store_lat = request.form.get("store_lat") or None
        db.session.commit()
        flash("门店已更新。", "success")
        return redirect(url_for("admin.stores"))
    return render_template("admin/store_form.html", store=store, provinces=provinces)
```

- **要点**：门店创建与编辑时同步冗余保存 `province_name`，列表展示无需额外联表查询。
- **要点**：经纬度等可选字段以空字符串转 `None` 的方式入库，保持空值语义一致。
- **要点**：省份/区域与门店的全部路由均受 `store:manage` 权限控制，与用户管理权限相互独立。

## 阶段小结

| 知识模块 | 核心内容 | 在本阶段中的作用 |
|----------|----------|------------------|
| 用户管理路由 | 关键词搜索、新增、编辑、状态禁用 | 提供账号与人员数据的基础维护能力 |

| 省份/区域管理 | StoreProvince 的新增与编辑 | 为门店提供区域归属数据 |

| 用户绑定门店 | UserStore 中间表，覆盖式重建关联 | 实现用户与门店的多对多关系 |
| 权限菜单 | `has_permission` 包裹菜单入口 | 与 `permission_required` 构成双重校验 |

## 常见问题与排查

| 现象 | 原因 | 处理方式 |
|------|------|----------|
| 访问 `/admin/users` 返回 404 | `user_routes` 未在蓝图 `__init__.py` 中导入注册 | 补充 `from . import user_routes` 并重启开发服务器 |
| 菜单不显示用户或门店入口 | 当前角色未分配 `user:manage` 或 `store:manage` 权限 | 由超级管理员为角色分配权限后重新登录 |
| 编辑用户时档案字段未保存 | 该用户缺少 UserProfile 记录，或表单处于新增态未渲染档案字段 | 档案字段仅在编辑态提交，确认对已有用户执行编辑操作 |
| 门店绑定保存后勾选全部丢失 | 复选框 `name` 与服务端读取的 `store_ids` 不一致，或请求未以 POST 提交 | 核对表单字段名为 `store_ids`，确认路由 methods 包含 POST |
| 新增门店提交后返回 404 | 提交的 `province_id` 不存在，被 `get_or_404` 拦截 | 先创建省份/区域，再在门店表单中选择有效区域 |

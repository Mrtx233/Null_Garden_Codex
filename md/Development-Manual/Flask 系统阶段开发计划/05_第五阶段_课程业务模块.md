# 第 5 阶段：课程业务模块

本阶段基于第 4 阶段用户与门店基础模块继续开发，实现课程分类管理、课程管理、门店课程排期管理和课程收藏功能。

完成本阶段后：

- 后台可以维护课程分类。
- 后台可以维护课程主数据。
- 后台可以给门店安排课程排期。
- 前台可以查看课程列表和课程详情。
- 登录用户可以收藏或取消收藏课程。

## 3. 课程后台路由

文件：`app/blueprints/admin/course_routes.py`

```python
from flask import flash, redirect, render_template, request, url_for

from app.extensions import db
from app.models import Course, CourseCategory, Store, StoreCourseSchedule, User
from app.utils.auth import permission_required
from . import admin_bp


@admin_bp.route("/course-categories")
@permission_required("course:manage")
def course_categories():
    items = CourseCategory.query.order_by(CourseCategory.category_id.desc()).all()
    return render_template("admin/course_categories.html", categories=items)


@admin_bp.route("/course-categories/create", methods=["GET", "POST"])
@permission_required("course:manage")
def course_category_create():
    if request.method == "POST":
        item = CourseCategory(
            category_name=request.form.get("category_name", "").strip(),
            category_url=request.form.get("category_url", "").strip(),
            description=request.form.get("description", "").strip(),
            status=int(request.form.get("status", 1)),
        )
        db.session.add(item)
        db.session.commit()
        flash("课程分类已创建。", "success")
        return redirect(url_for("admin.course_categories"))
    return render_template("admin/course_category_form.html", category=None)


@admin_bp.route("/course-categories/<int:category_id>/edit", methods=["GET", "POST"])
@permission_required("course:manage")
def course_category_edit(category_id):
    item = CourseCategory.query.get_or_404(category_id)
    if request.method == "POST":
        item.category_name = request.form.get("category_name", "").strip()
        item.category_url = request.form.get("category_url", "").strip()
        item.description = request.form.get("description", "").strip()
        item.status = int(request.form.get("status", 1))
        db.session.commit()
        flash("课程分类已更新。", "success")
        return redirect(url_for("admin.course_categories"))
    return render_template("admin/course_category_form.html", category=item)


@admin_bp.route("/courses")
@permission_required("course:manage")
def courses():
    items = Course.query.order_by(Course.course_id.desc()).all()
    return render_template("admin/courses.html", courses=items)


@admin_bp.route("/courses/create", methods=["GET", "POST"])
@permission_required("course:manage")
def course_create():
    categories = CourseCategory.query.order_by(CourseCategory.category_id.asc()).all()
    if request.method == "POST":
        item = Course(
            course_name=request.form.get("course_name", "").strip(),
            category_id=int(request.form["category_id"]),
            course_difficulty=request.form.get("course_difficulty") or None,
            duration_minutes=request.form.get("duration_minutes") or None,
            max_participants=request.form.get("max_participants") or None,
            schedule_info=request.form.get("schedule_info", "").strip(),
            description=request.form.get("description", "").strip(),
            status=int(request.form.get("status", 1)),
        )
        db.session.add(item)
        db.session.commit()
        flash("课程已创建。", "success")
        return redirect(url_for("admin.courses"))
    return render_template("admin/course_form.html", course=None, categories=categories)


@admin_bp.route("/courses/<int:course_id>/edit", methods=["GET", "POST"])
@permission_required("course:manage")
def course_edit(course_id):
    item = Course.query.get_or_404(course_id)
    categories = CourseCategory.query.order_by(CourseCategory.category_id.asc()).all()
    if request.method == "POST":
        item.course_name = request.form.get("course_name", "").strip()
        item.category_id = int(request.form["category_id"])
        item.course_difficulty = request.form.get("course_difficulty") or None
        item.duration_minutes = request.form.get("duration_minutes") or None
        item.max_participants = request.form.get("max_participants") or None
        item.schedule_info = request.form.get("schedule_info", "").strip()
        item.description = request.form.get("description", "").strip()
        item.status = int(request.form.get("status", 1))
        db.session.commit()
        flash("课程已更新。", "success")
        return redirect(url_for("admin.courses"))
    return render_template("admin/course_form.html", course=item, categories=categories)


@admin_bp.route("/course-schedules")
@permission_required("course:manage")
def course_schedules():
    items = StoreCourseSchedule.query.order_by(StoreCourseSchedule.start_time.desc()).all()
    return render_template("admin/course_schedules.html", schedules=items)


@admin_bp.route("/course-schedules/create", methods=["GET", "POST"])
@permission_required("course:manage")
def course_schedule_create():
    stores = Store.query.order_by(Store.store_id.asc()).all()
    courses = Course.query.order_by(Course.course_id.asc()).all()
    coaches = User.query.order_by(User.user_id.asc()).all()
    if request.method == "POST":
        item = StoreCourseSchedule(
            store_id=int(request.form["store_id"]),
            course_id=int(request.form["course_id"]),
            coach_id=request.form.get("coach_id") or None,
            course_date=request.form["course_date"],
            start_time=request.form["start_time"],
            classroom_name=request.form.get("classroom_name", "").strip(),
            status=int(request.form.get("status", 1)),
        )
        db.session.add(item)
        db.session.commit()
        flash("课程排期已创建。", "success")
        return redirect(url_for("admin.course_schedules"))
    return render_template("admin/course_schedule_form.html", schedule=None, stores=stores, courses=courses, coaches=coaches)
```

## 4. 前台课程路由和收藏

修改文件：`app/blueprints/web/routes.py`

```python
from flask import flash, redirect, render_template, url_for

from app.extensions import db
from app.models import Course, CourseCategory, UserCourseFavorite
from app.utils.auth import get_current_user, login_required
from . import web_bp


@web_bp.route("/")
def index():
    courses = Course.query.filter_by(status=1).order_by(Course.course_id.desc()).limit(6).all()
    return render_template("web/index.html", courses=courses)


@web_bp.route("/courses")
def course_list():
    categories = CourseCategory.query.filter_by(status=1).order_by(CourseCategory.category_id.asc()).all()
    courses = Course.query.filter_by(status=1).order_by(Course.course_id.desc()).all()
    return render_template("web/courses.html", categories=categories, courses=courses)


@web_bp.route("/courses/<int:course_id>")
def course_detail(course_id):
    course = Course.query.get_or_404(course_id)
    user = get_current_user()
    is_favorite = False
    if user:
        is_favorite = UserCourseFavorite.query.filter_by(user_id=user.user_id, course_id=course.course_id).first() is not None
    return render_template("web/course_detail.html", course=course, is_favorite=is_favorite)


@web_bp.route("/courses/<int:course_id>/favorite", methods=["POST"])
@login_required
def course_favorite(course_id):
    user = get_current_user()
    favorite = UserCourseFavorite.query.filter_by(user_id=user.user_id, course_id=course_id).first()
    if favorite:
        db.session.delete(favorite)
        flash("已取消收藏课程。", "success")
    else:
        db.session.add(UserCourseFavorite(user_id=user.user_id, course_id=course_id))
        flash("已收藏课程。", "success")
    db.session.commit()
    return redirect(url_for("web.course_detail", course_id=course_id))
```

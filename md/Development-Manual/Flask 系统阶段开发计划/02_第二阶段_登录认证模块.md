# 第 2 阶段：登录认证模块

本阶段基于第 1 阶段项目骨架继续开发，实现后台账号密码登录、退出登录、Session 保存登录状态、后台访问拦截和初始化管理员账号。

完成本阶段后：

- `/auth/login` 可以登录后台。
- `/auth/logout` 可以退出登录。
- `/admin/` 必须登录后访问。
- Session 中保存 `user_id`。
- 密码使用 Werkzeug 哈希，不保存明文。

```python
from functools import wraps

from flask import flash, redirect, request, session, url_for

from app.models import User


def get_current_user():
    user_id = session.get("user_id")
    if not user_id:
        return None
    return User.query.get(user_id)


def login_required(view_func):
    @wraps(view_func)
    def wrapper(*args, **kwargs):
        user = get_current_user()
        if not user:
            flash("请先登录后再访问后台。", "warning")
            return redirect(url_for("auth.login", next=request.full_path))
        if not user.is_active:
            session.clear()
            flash("当前账号已被禁用。", "danger")
            return redirect(url_for("auth.login"))
        return view_func(*args, **kwargs)

    return wrapper
```

```python
import os

from flask import Flask, render_template

from .extensions import db, migrate

    app = Flask(__name__)

    env_name = config_name or os.getenv("FLASK_ENV", "development")

    db.init_app(app)
    migrate.init_app(app, db)


def register_blueprints(app):
    from .blueprints.admin import admin_bp
    from .blueprints.auth import auth_bp
    from .blueprints.web import web_bp

    app.register_blueprint(web_bp)
    app.register_blueprint(auth_bp, url_prefix="/auth")
    app.register_blueprint(admin_bp, url_prefix="/admin")


def register_commands(app):
    from .commands import register_cli_commands

    register_cli_commands(app)


def register_template_context(app):
    from .utils.auth import get_current_user

    @app.context_processor
    def inject_current_user():
        return {"current_user": get_current_user()}


def register_error_handlers(app):
    @app.errorhandler(404)
    def page_not_found(error):
        return render_template("errors/404.html"), 404

    @app.errorhandler(500)
    def internal_server_error(error):
        return render_template("errors/500.html"), 500
```

## 4. 初始化管理员命令

文件：`app/commands.py`

```python
import click
from flask.cli import with_appcontext

from app.extensions import db
from app.models import Role, User, UserProfile


def register_cli_commands(app):
    app.cli.add_command(init_admin)


@click.command("init-admin")
@click.option("--username", default="admin", help="管理员用户名")
@click.option("--password", default="123456", help="管理员密码")
@with_appcontext
def init_admin(username, password):
    admin_role = Role.query.filter_by(role_code="admin").first()
    if not admin_role:
        admin_role = Role(role_code="admin", role_name="超级管理员")
        db.session.add(admin_role)
        db.session.flush()

    user = User.query.filter_by(username=username).first()
    if not user:
        user = User(
            role_id=admin_role.id,
            username=username,
            real_name="系统管理员",
            phone="",
            status=1,
        )
        user.set_password(password)
        db.session.add(user)
        db.session.flush()

        profile = UserProfile(user_id=user.user_id)
        db.session.add(profile)
        click.echo(f"已创建管理员账号：{username}")
    else:
        user.role_id = admin_role.id
        user.status = 1
        user.set_password(password)
        click.echo(f"已重置管理员账号：{username}")

    db.session.commit()
    click.echo("初始化完成。")
```

执行：


## 5. 登录和退出路由

修改文件：`app/blueprints/auth/routes.py`

```python
from flask import flash, redirect, render_template, request, session, url_for

from app.models import User
from . import auth_bp


@auth_bp.route("/login", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        username = request.form.get("username", "").strip()
        password = request.form.get("password", "")
        next_url = request.args.get("next") or url_for("admin.dashboard")

        user = User.query.filter_by(username=username).first()
        if not user or not user.check_password(password):
            flash("用户名或密码错误。", "danger")
            return render_template("auth/login.html")

        if not user.is_active:
            flash("当前账号已被禁用，请联系管理员。", "danger")
            return render_template("auth/login.html")

        session.clear()
        session["user_id"] = user.user_id
        session["username"] = user.username
        flash("登录成功。", "success")
        return redirect(next_url)

    return render_template("auth/login.html")


@auth_bp.route("/logout")
def logout():
    session.clear()
    flash("已退出登录。", "success")
    return redirect(url_for("auth.login"))
```

## 3. 动作库后台路由

文件：`app/blueprints/admin/action_routes.py`

```python
from flask import flash, redirect, render_template, request, url_for

from app.extensions import db
from app.models import Action, ActionCategory
from app.utils.auth import permission_required
from . import admin_bp


@admin_bp.route("/action-categories")
@permission_required("action:manage")
def action_categories():
    items = ActionCategory.query.order_by(ActionCategory.category_id.desc()).all()
    return render_template("admin/action_categories.html", categories=items)


@admin_bp.route("/action-categories/create", methods=["GET", "POST"])
@permission_required("action:manage")
def action_category_create():
    if request.method == "POST":
        item = ActionCategory(
            category_name=request.form.get("category_name", "").strip(),
            category_image_url=request.form.get("category_image_url", "").strip(),
        )
        db.session.add(item)
        db.session.commit()
        flash("动作分类已创建。", "success")
        return redirect(url_for("admin.action_categories"))
    return render_template("admin/action_category_form.html", category=None)


@admin_bp.route("/action-categories/<int:category_id>/edit", methods=["GET", "POST"])
@permission_required("action:manage")
def action_category_edit(category_id):
    item = ActionCategory.query.get_or_404(category_id)
    if request.method == "POST":
        item.category_name = request.form.get("category_name", "").strip()
        item.category_image_url = request.form.get("category_image_url", "").strip()
        db.session.commit()
        flash("动作分类已更新。", "success")
        return redirect(url_for("admin.action_categories"))
    return render_template("admin/action_category_form.html", category=item)


@admin_bp.route("/actions")
@permission_required("action:manage")
def actions():
    items = Action.query.order_by(Action.action_id.desc()).all()
    return render_template("admin/actions.html", actions=items)


@admin_bp.route("/actions/create", methods=["GET", "POST"])
@permission_required("action:manage")
def action_create():
    categories = ActionCategory.query.order_by(ActionCategory.category_id.asc()).all()
    if request.method == "POST":
        item = Action(
            action_name=request.form.get("action_name", "").strip(),
            category_id=int(request.form["category_id"]),
            action_difficulty=request.form.get("action_difficulty") or None,
            action_image_url=request.form.get("action_image_url", "").strip(),
            action_steps=request.form.get("action_steps", "").strip(),
            attention_points=request.form.get("attention_points", "").strip(),
            applicable_equipment=request.form.get("applicable_equipment", "").strip(),
            applicable_store_type=request.form.get("applicable_store_type") or None,
        )
        db.session.add(item)
        db.session.commit()
        flash("动作已创建。", "success")
        return redirect(url_for("admin.actions"))
    return render_template("admin/action_form.html", action=None, categories=categories)


@admin_bp.route("/actions/<int:action_id>/edit", methods=["GET", "POST"])
@permission_required("action:manage")
def action_edit(action_id):
    item = Action.query.get_or_404(action_id)
    categories = ActionCategory.query.order_by(ActionCategory.category_id.asc()).all()
    if request.method == "POST":
        item.action_name = request.form.get("action_name", "").strip()
        item.category_id = int(request.form["category_id"])
        item.action_difficulty = request.form.get("action_difficulty") or None
        item.action_image_url = request.form.get("action_image_url", "").strip()
        item.action_steps = request.form.get("action_steps", "").strip()
        item.attention_points = request.form.get("attention_points", "").strip()
        item.applicable_equipment = request.form.get("applicable_equipment", "").strip()
        item.applicable_store_type = request.form.get("applicable_store_type") or None
        db.session.commit()
        flash("动作已更新。", "success")
        return redirect(url_for("admin.actions"))
    return render_template("admin/action_form.html", action=item, categories=categories)
```

## 4. 前台动作库路由和收藏

在第 5 阶段 `app/blueprints/web/routes.py` 的基础上继续增加以下路由。

```python
from app.models import Action, ActionCategory, UserActionFavorite


@web_bp.route("/actions")
def action_list():
    categories = ActionCategory.query.order_by(ActionCategory.category_id.asc()).all()
    actions = Action.query.order_by(Action.action_id.desc()).all()
    return render_template("web/actions.html", categories=categories, actions=actions)


@web_bp.route("/actions/<int:action_id>")
def action_detail(action_id):
    action = Action.query.get_or_404(action_id)
    user = get_current_user()
    is_favorite = False
    if user:
        is_favorite = UserActionFavorite.query.filter_by(user_id=user.user_id, action_id=action.action_id).first() is not None
    return render_template("web/action_detail.html", action=action, is_favorite=is_favorite)


@web_bp.route("/actions/<int:action_id>/favorite", methods=["POST"])
@login_required
def action_favorite(action_id):
    user = get_current_user()
    favorite = UserActionFavorite.query.filter_by(user_id=user.user_id, action_id=action_id).first()
    if favorite:
        db.session.delete(favorite)
        flash("已取消收藏动作。", "success")
    else:
        db.session.add(UserActionFavorite(user_id=user.user_id, action_id=action_id))
        flash("已收藏动作。", "success")
    db.session.commit()
    return redirect(url_for("web.action_detail", action_id=action_id))
```

# 05_后端课程模块

## 一、阶段目标

本阶段目标是在用户、门店、RBAC 权限已经完成的基础上，实现课程分类、课程主数据、用户课程收藏等后端接口。

完成本阶段后，后端应具备以下能力：

1. 管理员可以维护课程分类。
2. 管理员可以维护课程主数据。
3. 管理员可以按分类、难度、状态、关键字筛选课程。
4. 登录用户可以收藏课程。
5. 登录用户可以取消收藏课程。
6. 登录用户可以查看自己的课程收藏列表。
7. 后台管理接口接入 `course:*` 权限。
8. 用户收藏接口只要求登录，不额外要求后台权限。

## 二、本阶段开发顺序

```text
1. 创建课程 Schema
2. 创建课程 Service
3. 创建课程 API 路由
4. 注册课程路由

6. 启动服务并在 Swagger 验证接口
```

    max_participants: int | None = Field(default=None, ge=1)

    course_name: str

    duration_minutes: int | None = None
    max_participants: int | None = None

### 注意事项

### 相对路径

```text
backend/app/services/course.py
```

### 完整代码

```python
from sqlalchemy import func, select
from sqlalchemy.orm import Session

from app.core.exceptions import BusinessException
from app.models.course import Course, CourseCategory, UserCourseFavorite
from app.schemas.course import CourseCategoryCreate, CourseCategoryUpdate, CourseCreate, CourseUpdate
from app.utils.pagination import paginate


def list_course_categories(db: Session) -> list[CourseCategory]:
    stmt = select(CourseCategory).order_by(CourseCategory.category_id.asc())
    return list(db.scalars(stmt).all())


def get_course_category_or_404(db: Session, category_id: int) -> CourseCategory:
    category = db.get(CourseCategory, category_id)
    if not category:
        raise BusinessException(message="course category not found", code=404)
    return category


def create_course_category(
    db: Session,
    category_in: CourseCategoryCreate,
) -> CourseCategory:
    category = CourseCategory(**category_in.model_dump())
    db.add(category)
    db.commit()
    db.refresh(category)
    return category


def update_course_category(
    db: Session,
    category_id: int,
    category_in: CourseCategoryUpdate,
) -> CourseCategory:
    category = get_course_category_or_404(db, category_id)
    update_data = category_in.model_dump(exclude_unset=True)

    for field, value in update_data.items():
        setattr(category, field, value)

    db.add(category)
    db.commit()
    db.refresh(category)
    return category


def delete_course_category(db: Session, category_id: int) -> None:
    category = get_course_category_or_404(db, category_id)
    course_count = db.scalar(
        select(func.count()).select_from(Course).where(Course.category_id == category_id)
    )
    if course_count:
        raise BusinessException(message="category has courses, cannot delete", code=400)

    db.delete(category)
    db.commit()


def list_courses(
    db: Session,
    page: int = 1,
    page_size: int = 10,
    keyword: str | None = None,
    category_id: int | None = None,
    course_difficulty: int | None = None,
    status: int | None = None,
):
    stmt = select(Course).order_by(Course.course_id.desc())

    if keyword:
        stmt = stmt.where(Course.course_name.like(f"%{keyword}%"))
    if category_id is not None:
        stmt = stmt.where(Course.category_id == category_id)
    if course_difficulty is not None:
        stmt = stmt.where(Course.course_difficulty == course_difficulty)
    if status is not None:
        stmt = stmt.where(Course.status == status)

    return paginate(db, stmt, page=page, page_size=page_size)


def get_course_or_404(db: Session, course_id: int) -> Course:
    course = db.get(Course, course_id)
    if not course:
        raise BusinessException(message="course not found", code=404)
    return course


def create_course(db: Session, course_in: CourseCreate) -> Course:
    get_course_category_or_404(db, course_in.category_id)

    course = Course(**course_in.model_dump())
    db.add(course)
    db.commit()
    db.refresh(course)
    return course


def update_course(
    db: Session,
    course_id: int,
    course_in: CourseUpdate,
) -> Course:
    course = get_course_or_404(db, course_id)
    update_data = course_in.model_dump(exclude_unset=True)

    if "category_id" in update_data:
        get_course_category_or_404(db, update_data["category_id"])

    for field, value in update_data.items():
        setattr(course, field, value)

    db.add(course)
    db.commit()
    db.refresh(course)
    return course


def delete_course(db: Session, course_id: int) -> None:
    course = get_course_or_404(db, course_id)
    favorite_count = db.scalar(
        select(func.count()).select_from(UserCourseFavorite).where(
            UserCourseFavorite.course_id == course_id
        )
    )
    if favorite_count:
        raise BusinessException(message="course has favorites, cannot delete", code=400)

    db.delete(course)
    db.commit()


def favorite_course(
    db: Session,
    user_id: int,
    course_id: int,
) -> UserCourseFavorite:
    get_course_or_404(db, course_id)

    favorite = db.scalar(
        select(UserCourseFavorite).where(
            UserCourseFavorite.user_id == user_id,
            UserCourseFavorite.course_id == course_id,
        )
    )
    if favorite:
        return favorite

    favorite = UserCourseFavorite(user_id=user_id, course_id=course_id)
    db.add(favorite)
    db.commit()
    db.refresh(favorite)
    return favorite


def unfavorite_course(
    db: Session,
    user_id: int,
    course_id: int,
) -> None:
    favorite = db.scalar(
        select(UserCourseFavorite).where(
            UserCourseFavorite.user_id == user_id,
            UserCourseFavorite.course_id == course_id,
        )
    )
    if not favorite:
        raise BusinessException(message="course favorite not found", code=404)

    db.delete(favorite)
    db.commit()


def list_my_course_favorites(db: Session, user_id: int) -> list[UserCourseFavorite]:
    stmt = (
        select(UserCourseFavorite)
        .where(UserCourseFavorite.user_id == user_id)
        .order_by(UserCourseFavorite.favorite_id.desc())
    )
    return list(db.scalars(stmt).all())
```

### 注意事项

1. 删除分类前检查是否有课程。
2. 删除课程前检查是否被收藏。
3. 收藏接口重复调用不会重复插入，直接返回已有收藏。
4. 取消收藏时如果不存在，返回业务错误。

from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session

from app.api.deps import get_current_user, require_permission
from app.db.session import get_db
from app.models.course import Course, CourseCategory, UserCourseFavorite
from app.models.user import SysUser
from app.schemas.course import (
    CourseCategoryCreate,
    CourseCategoryUpdate,

from app.services.course import (

    create_course_category,
    delete_course,
    delete_course_category,

    list_courses,

    unfavorite_course,
    update_course,
    update_course_category,

from app.utils.response import success


router = APIRouter(tags=["课程管理"])


def category_to_dict(item: CourseCategory) -> dict:
    return {
        "category_id": item.category_id,
        "category_name": item.category_name,
        "category_url": item.category_url,
        "description": item.description,
        "status": item.status,
    }


def course_to_dict(item: Course) -> dict:
    return {
        "course_id": item.course_id,
        "course_name": item.course_name,
        "category_id": item.category_id,
        "course_difficulty": item.course_difficulty,
        "duration_minutes": item.duration_minutes,
        "max_participants": item.max_participants,
        "schedule_info": item.schedule_info,
        "description": item.description,
        "status": item.status,
    }


def favorite_to_dict(item: UserCourseFavorite) -> dict:
    return {
        "favorite_id": item.favorite_id,
        "user_id": item.user_id,
        "course_id": item.course_id,
        "created_at": item.created_at,
    }


@router.get("/course-categories")
def get_course_categories(
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:list")),
) -> dict:
    items = list_course_categories(db)
    return success(data=[category_to_dict(item) for item in items])


@router.post("/course-categories")
def add_course_category(
    payload: CourseCategoryCreate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:create")),
) -> dict:
    category = create_course_category(db, payload)
    return success(data=category_to_dict(category))


@router.put("/course-categories/{category_id}")
def edit_course_category(
    category_id: int,
    payload: CourseCategoryUpdate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:update")),
) -> dict:
    category = update_course_category(db, category_id, payload)
    return success(data=category_to_dict(category))


@router.delete("/course-categories/{category_id}")
def remove_course_category(
    category_id: int,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:delete")),
) -> dict:
    delete_course_category(db, category_id)
    return success(message="deleted")


@router.get("/courses")
def get_courses(
    page: int = Query(default=1, ge=1),
    page_size: int = Query(default=10, ge=1, le=100),
    keyword: str | None = Query(default=None),
    category_id: int | None = Query(default=None),
    course_difficulty: int | None = Query(default=None),
    status: int | None = Query(default=None),
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:list")),
) -> dict:
    result = list_courses(
        db,
        page=page,
        page_size=page_size,
        keyword=keyword,
        category_id=category_id,
        course_difficulty=course_difficulty,
        status=status,
    )
    return success(
        data={
            "items": [course_to_dict(item) for item in result.items],
            "total": result.total,
            "page": result.page,
            "page_size": result.page_size,
            "pages": result.pages,
        }
    )


@router.post("/courses")
def add_course(
    payload: CourseCreate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:create")),
) -> dict:
    course = create_course(db, payload)
    return success(data=course_to_dict(course))


@router.get("/courses/{course_id}")
def get_course_detail(
    course_id: int,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:list")),
) -> dict:
    course = get_course_or_404(db, course_id)
    return success(data=course_to_dict(course))


@router.put("/courses/{course_id}")
def edit_course(
    course_id: int,
    payload: CourseUpdate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:update")),
) -> dict:
    course = update_course(db, course_id, payload)
    return success(data=course_to_dict(course))


@router.delete("/courses/{course_id}")
def remove_course(
    course_id: int,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("course:delete")),
) -> dict:
    delete_course(db, course_id)

@router.post("/courses/{course_id}/favorite")

## 十三、常见问题

### 1. 创建课程提示 course category not found

原因：

传入的 `category_id` 不存在。

修复：

先调用：

```text
POST /api/course-categories
```

创建课程分类，再使用返回的 `category_id` 创建课程。

### 2. 删除课程分类提示 category has courses, cannot delete

原因：

该分类下已经有课程。

修复：

先删除或迁移该分类下的课程。

### 3. 删除课程提示 course has favorites, cannot delete

原因：

课程已经被用户收藏。

修复：

先取消相关收藏，或改成后台逻辑删除课程。

### 4. 收藏课程提示 course not found

原因：

路径中的 `course_id` 不存在。

修复：

先调用：

```text
GET /api/courses
```

找到真实存在的 `course_id`。

### 5. 后台课程接口返回 403 permission denied

原因：

当前用户角色没有 `course:list`、`course:create`、`course:update` 或 `course:delete` 权限。

修复：

使用管理员账号测试，或给当前角色分配课程权限。

# 07_后端活动首页模块

## 一、阶段目标

本阶段目标是在动作库模块已经完成的基础上，实现 Slogan 标语、赛事活动、首页聚合数据等后端接口。

完成本阶段后，后端应具备以下能力：

1. 管理员可以维护首页 Slogan 标语。
2. 管理员可以维护赛事与活动信息。
3. 管理员可以按状态、关键字、日期筛选活动。
4. 前端首页可以获取启用状态的 Slogan。
5. 前端首页可以获取活动列表摘要。
6. 后台管理接口接入 `slogan:*` 和 `event:*` 权限。
7. 首页聚合接口不要求登录，方便前端首页直接展示。

## 二、本阶段开发顺序

```text
1. 补充 Slogan 权限种子
2. 创建内容活动 Schema
3. 创建内容活动 Service
4. 创建内容活动 API 路由
5. 注册内容活动路由
6. 修改路由注册测试
7. 启动服务并在 Swagger 验证接口
```

```python
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.core.config import settings
from app.core.security import get_password_hash
from app.models.system import SysPermission, SysRole, SysRolePermission
from app.models.user import SysUser


logger = logging.getLogger(__name__)


DEFAULT_ROLES = [
    ("admin", "超级管理员"),
    ("store_admin", "门店管理员"),
    ("coach", "教练"),
    ("member", "普通会员"),

    ("store:update", "修改门店", None),
    ("store:delete", "删除门店", None),
    ("course:list", "课程列表", "/courses"),
    ("course:create", "新增课程", None),
    ("course:update", "修改课程", None),
    ("course:delete", "删除课程", None),
    ("action:list", "动作列表", "/actions"),
    ("action:create", "新增动作", None),

    ("action:delete", "删除动作", None),
    ("event:list", "活动列表", "/content/events"),
    ("event:create", "新增活动", None),
    ("event:update", "修改活动", None),
    ("event:delete", "删除活动", None),
    ("slogan:list", "Slogan列表", "/content/slogans"),
    ("slogan:create", "新增Slogan", None),
    ("slogan:update", "修改Slogan", None),
    ("slogan:delete", "删除Slogan", None),
]


def get_or_create_role(db: Session, role_code: str, role_name: str) -> SysRole:
    role = db.scalar(select(SysRole).where(SysRole.role_code == role_code))
    if role:
        return role

    role = SysRole(role_code=role_code, role_name=role_name)
    db.add(role)
    db.flush()
    return role


def get_or_create_permission(
    db: Session,
    permission_code: str,
    permission_name: str,
    menu_path: str | None,
) -> SysPermission:
    permission = db.scalar(
        select(SysPermission).where(SysPermission.permission_code == permission_code)
    )
    if permission:
        return permission

    permission = SysPermission(
        permission_code=permission_code,
        permission_name=permission_name,
        menu_path=menu_path,
    )
    db.add(permission)
    db.flush()
    return permission


def bind_role_permission(
    db: Session,
    role_id: int,
    permission_id: int,
) -> None:
    exists = db.scalar(
        select(SysRolePermission).where(
            SysRolePermission.role_id == role_id,
            SysRolePermission.permission_id == permission_id,
        )
    )
    if exists:
        return

    db.add(
        SysRolePermission(
            role_id=role_id,
            permission_id=permission_id,
        )
    )


def seed_roles_and_permissions(db: Session) -> None:
    roles = {
        role_code: get_or_create_role(db, role_code, role_name)
        for role_code, role_name in DEFAULT_ROLES
    }

    permissions = [
        get_or_create_permission(db, code, name, menu_path)
        for code, name, menu_path in DEFAULT_PERMISSIONS
    ]

    admin_role = roles["admin"]
    for permission in permissions:
        bind_role_permission(db, admin_role.id, permission.id)


def seed_admin_user(db: Session) -> None:
    admin_user = db.scalar(
        select(SysUser).where(SysUser.username == settings.DEFAULT_ADMIN_USERNAME)
    )
    if admin_user:
        return

    admin_role = db.scalar(select(SysRole).where(SysRole.role_code == "admin"))
    if not admin_role:
        raise RuntimeError("admin role is missing")

    admin_user = SysUser(
        role_id=admin_role.id,
        username=settings.DEFAULT_ADMIN_USERNAME,
        password=get_password_hash(settings.DEFAULT_ADMIN_PASSWORD),
        real_name=settings.DEFAULT_ADMIN_REAL_NAME,
        status=1,
    )
    db.add(admin_user)


def seed_initial_data(db: Session) -> None:
    seed_roles_and_permissions(db)
    seed_admin_user(db)
    db.commit()
    logger.info("Initial data checked")
```

### 注意事项

1. 本阶段新增 `slogan:create` 和 `slogan:delete`，用于让 Slogan 管理权限更完整。
2. 如果数据库已经存在，重启后端时会自动补齐缺失权限并绑定给 `admin`。
3. 如果你关闭了 `AUTO_SEED_DATA`，需要先打开一次并重启服务，或者手动插入权限数据。

### 完整代码

```python
from datetime import date, datetime

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.core.exceptions import BusinessException
from app.models.content import ActivityEvent, SloganInfo
from app.schemas.content import (
    ActivityEventCreate,
    ActivityEventUpdate,
    SloganCreate,
    SloganUpdate,
)
from app.utils.pagination import paginate


def list_slogans(db: Session, status: int | None = None) -> list[SloganInfo]:
    stmt = select(SloganInfo).order_by(SloganInfo.slogan_id.asc())

    if status is not None:
        stmt = stmt.where(SloganInfo.status == status)

    return list(db.scalars(stmt).all())


def get_slogan_or_404(db: Session, slogan_id: int) -> SloganInfo:
    slogan = db.get(SloganInfo, slogan_id)
    if not slogan:
        raise BusinessException(message="slogan not found", code=404)
    return slogan


def create_slogan(db: Session, slogan_in: SloganCreate) -> SloganInfo:
    slogan = SloganInfo(**slogan_in.model_dump())
    db.add(slogan)
    db.commit()
    db.refresh(slogan)
    return slogan


def update_slogan(
    db: Session,
    slogan_id: int,
    slogan_in: SloganUpdate,
) -> SloganInfo:
    slogan = get_slogan_or_404(db, slogan_id)
    update_data = slogan_in.model_dump(exclude_unset=True)

    for field, value in update_data.items():
        setattr(slogan, field, value)

    db.add(slogan)
    db.commit()
    db.refresh(slogan)
    return slogan


def delete_slogan(db: Session, slogan_id: int) -> None:
    slogan = get_slogan_or_404(db, slogan_id)
    db.delete(slogan)
    db.commit()


def list_activity_events(
    db: Session,
    page: int = 1,
    page_size: int = 10,
    keyword: str | None = None,
    status: str | None = None,
    date_from: date | None = None,
    date_to: date | None = None,
):
    stmt = select(ActivityEvent).order_by(
        ActivityEvent.event_date.desc(),
        ActivityEvent.event_id.desc(),
    )

    if keyword:
        stmt = stmt.where(ActivityEvent.title.like(f"%{keyword}%"))
    if status:
        stmt = stmt.where(ActivityEvent.status == status)
    if date_from is not None:
        stmt = stmt.where(ActivityEvent.event_date >= date_from)
    if date_to is not None:
        stmt = stmt.where(ActivityEvent.event_date <= date_to)

    return paginate(db, stmt, page=page, page_size=page_size)


def get_activity_event_or_404(db: Session, event_id: int) -> ActivityEvent:
    event = db.get(ActivityEvent, event_id)
    if not event:
        raise BusinessException(message="activity event not found", code=404)
    return event


def create_activity_event(
    db: Session,
    event_in: ActivityEventCreate,
) -> ActivityEvent:
    event = ActivityEvent(**event_in.model_dump())
    db.add(event)
    db.commit()
    db.refresh(event)
    return event


def update_activity_event(
    db: Session,
    event_id: int,
    event_in: ActivityEventUpdate,
) -> ActivityEvent:
    event = get_activity_event_or_404(db, event_id)
    update_data = event_in.model_dump(exclude_unset=True)

    for field, value in update_data.items():
        setattr(event, field, value)

    event.updated_at = datetime.now()
    db.add(event)
    db.commit()
    db.refresh(event)
    return event


def delete_activity_event(db: Session, event_id: int) -> None:
    event = get_activity_event_or_404(db, event_id)
    db.delete(event)
    db.commit()


def get_home_content(
    db: Session,
    slogan_limit: int = 5,
    event_limit: int = 6,
) -> dict:
    slogan_stmt = (
        select(SloganInfo)
        .where(SloganInfo.status == 1)
        .order_by(SloganInfo.slogan_id.asc())
        .limit(slogan_limit)
    )
    event_stmt = (
        select(ActivityEvent)
        .order_by(ActivityEvent.event_date.desc(), ActivityEvent.event_id.desc())
        .limit(event_limit)
    )

    return {
        "slogans": list(db.scalars(slogan_stmt).all()),
        "events": list(db.scalars(event_stmt).all()),
    }
```

### 注意事项

1. 首页只返回 `status=1` 的 Slogan。
2. 活动列表按活动日期倒序展示。
3. `updated_at` 在更新活动时由 Service 主动刷新，避免 ORM 自动建表时缺少 MySQL `ON UPDATE` 行为。

```python
from datetime import date

from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session

from app.api.deps import require_permission
from app.db.session import get_db
from app.models.content import ActivityEvent, SloganInfo
from app.models.user import SysUser
from app.schemas.content import (
    ActivityEventCreate,
    ActivityEventUpdate,
    SloganCreate,
    SloganUpdate,
)
from app.services.content import (
    create_activity_event,
    create_slogan,
    delete_activity_event,
    delete_slogan,
    get_activity_event_or_404,
    get_home_content,
    list_activity_events,
    list_slogans,
    update_activity_event,
    update_slogan,
)
from app.utils.response import success


router = APIRouter(tags=["内容与活动"])


def slogan_to_dict(item: SloganInfo) -> dict:
    return {
        "slogan_id": item.slogan_id,
        "slogan_name": item.slogan_name,
        "slogan_content": item.slogan_content,
        "slogan_image_url": item.slogan_image_url,
        "status": item.status,
    }


def event_to_dict(item: ActivityEvent) -> dict:
    return {
        "event_id": item.event_id,
        "title": item.title,
        "event_date": item.event_date,
        "location": item.location,
        "status": item.status,
        "description": item.description,
        "tags": item.tags,
        "prize": item.prize,
        "scale": item.scale,
        "created_at": item.created_at,
        "updated_at": item.updated_at,
    }


@router.get("/home")
def get_home(db: Session = Depends(get_db)) -> dict:
    content = get_home_content(db)
    return success(
        data={
            "slogans": [slogan_to_dict(item) for item in content["slogans"]],
            "events": [event_to_dict(item) for item in content["events"]],
        }
    )


@router.get("/content/slogans")
def get_slogans(
    status: int | None = Query(default=None, ge=0, le=1),
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("slogan:list")),
) -> dict:
    items = list_slogans(db, status=status)
    return success(data=[slogan_to_dict(item) for item in items])


@router.post("/content/slogans")
def add_slogan(
    payload: SloganCreate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("slogan:create")),
) -> dict:
    slogan = create_slogan(db, payload)
    return success(data=slogan_to_dict(slogan))


@router.put("/content/slogans/{slogan_id}")
def edit_slogan(
    slogan_id: int,
    payload: SloganUpdate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("slogan:update")),
) -> dict:
    slogan = update_slogan(db, slogan_id, payload)
    return success(data=slogan_to_dict(slogan))


@router.delete("/content/slogans/{slogan_id}")
def remove_slogan(
    slogan_id: int,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("slogan:delete")),
) -> dict:
    delete_slogan(db, slogan_id)
    return success(message="deleted")


@router.get("/content/events")
def get_events(
    page: int = Query(default=1, ge=1),
    page_size: int = Query(default=10, ge=1, le=100),
    keyword: str | None = Query(default=None),
    status: str | None = Query(default=None, max_length=32),
    date_from: date | None = Query(default=None),
    date_to: date | None = Query(default=None),
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("event:list")),
) -> dict:
    result = list_activity_events(
        db,
        page=page,
        page_size=page_size,
        keyword=keyword,
        status=status,
        date_from=date_from,
        date_to=date_to,
    )
    return success(
        data={
            "items": [event_to_dict(item) for item in result.items],
            "total": result.total,
            "page": result.page,
            "page_size": result.page_size,
            "pages": result.pages,
        }
    )


@router.post("/content/events")
def add_event(
    payload: ActivityEventCreate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("event:create")),
) -> dict:
    event = create_activity_event(db, payload)
    return success(data=event_to_dict(event))


@router.get("/content/events/{event_id}")
def get_event_detail(
    event_id: int,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("event:list")),
) -> dict:
    event = get_activity_event_or_404(db, event_id)
    return success(data=event_to_dict(event))


@router.put("/content/events/{event_id}")
def edit_event(
    event_id: int,
    payload: ActivityEventUpdate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("event:update")),
) -> dict:
    event = update_activity_event(db, event_id, payload)
    return success(data=event_to_dict(event))


@router.delete("/content/events/{event_id}")
def remove_event(
    event_id: int,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("event:delete")),
) -> dict:
    delete_activity_event(db, event_id)
    return success(message="deleted")
```

### 注意事项

## 十五、常见问题

### 1. 创建 Slogan 返回 403 permission denied

原因：

数据库中缺少 `slogan:create` 权限，或者当前角色没有绑定该权限。

修复：

1. 确认 `.env` 中 `AUTO_SEED_DATA=true`。
2. 重启后端服务。
3. 使用管理员账号重新登录，获取新的 Token。
4. 再次调用创建 Slogan 接口。

### 2. 创建活动返回 422

原因：

请求体字段格式不符合 Schema，常见情况包括：

1. `event_date` 不是 `YYYY-MM-DD`。
2. `title`、`location`、`status`、`description` 为空。
3. `status` 超过 32 个字符。
4. `tags`、`prize`、`scale` 超过字段长度。

修复：

按照 Swagger 中的字段约束修改请求体。

### 3. 首页接口没有返回刚创建的 Slogan

原因：

首页接口只返回 `status=1` 的 Slogan。

修复：

确认 Slogan 的 `status` 是 `1`。

### 4. 首页接口返回 events 为空

原因：

当前数据库中还没有活动数据。

修复：

先调用：

```text
POST /api/content/events
```

创建活动，再访问：

```text
GET /api/home
```

### 5. 修改活动后 updated_at 没变化

原因：

如果只依赖 ORM 自动建表，MySQL 表结构不一定带有 `ON UPDATE CURRENT_TIMESTAMP`。

修复：

本阶段 `update_activity_event` 已经在 Service 中主动设置：

```python
event.updated_at = datetime.now()
```

如果仍然没变化，检查是否复制了完整的 `backend/app/services/content.py`。

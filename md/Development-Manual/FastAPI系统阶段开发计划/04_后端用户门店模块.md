# 04_后端用户门店模块

## 一、阶段目标

本阶段目标是在登录认证和 RBAC 权限已经完成的基础上，实现用户、用户档案、省份区域、门店、用户门店绑定等后端业务接口。

完成本阶段后，后端应具备以下能力：

1. 管理员可以分页查询用户。
2. 管理员可以创建用户，并自动加密密码。
3. 管理员可以修改用户基础信息。
4. 管理员可以启用或禁用用户。
5. 管理员可以查看和修改用户档案。
6. 管理员可以维护门店省份区域。
7. 管理员可以维护门店基础信息。
8. 管理员可以给用户绑定一个或多个门店。
9. 所有接口都接入 RBAC 权限校验。

## 二、本阶段开发顺序

```text
1. 创建用户相关 Schema
2. 创建门店相关 Schema
3. 创建用户 Service
4. 创建门店 Service

```

### 完整代码

```python
from datetime import datetime

from sqlalchemy import delete, or_, select
from sqlalchemy.orm import Session

from app.core.exceptions import BusinessException
from app.core.security import get_password_hash
from app.models.store import Store, UserStore
from app.models.system import SysRole
from app.models.user import SysUser, SysUserProfile
from app.schemas.user import UserCreate, UserProfileUpdate, UserStoreBind, UserUpdate
from app.utils.pagination import paginate


def list_users(
    db: Session,
    page: int = 1,
    page_size: int = 10,
    keyword: str | None = None,
    status: int | None = None,
    role_id: int | None = None,
):
    stmt = select(SysUser).order_by(SysUser.user_id.desc())

    if keyword:
        like_keyword = f"%{keyword}%"
        stmt = stmt.where(
            or_(
                SysUser.username.like(like_keyword),
                SysUser.real_name.like(like_keyword),
                SysUser.phone.like(like_keyword),
            )
        )
    if status is not None:
        stmt = stmt.where(SysUser.status == status)
    if role_id is not None:
        stmt = stmt.where(SysUser.role_id == role_id)

    return paginate(db, stmt, page=page, page_size=page_size)


def get_user_or_404(db: Session, user_id: int) -> SysUser:
    user = db.get(SysUser, user_id)
    if not user:
        raise BusinessException(message="user not found", code=404)
    return user


def ensure_role_exists(db: Session, role_id: int) -> None:
    role = db.get(SysRole, role_id)
    if not role:
        raise BusinessException(message="role not found", code=404)


def ensure_username_available(
    db: Session,
    username: str,
    exclude_user_id: int | None = None,
) -> None:
    stmt = select(SysUser).where(SysUser.username == username)
    if exclude_user_id is not None:
        stmt = stmt.where(SysUser.user_id != exclude_user_id)

    exists = db.scalar(stmt)
    if exists:
        raise BusinessException(message="username already exists", code=400)


def create_user(db: Session, user_in: UserCreate) -> SysUser:
    ensure_role_exists(db, user_in.role_id)
    ensure_username_available(db, user_in.username)

    user = SysUser(
        role_id=user_in.role_id,
        username=user_in.username,
        password=get_password_hash(user_in.password),
        real_name=user_in.real_name,
        phone=user_in.phone,
        status=user_in.status,
    )
    db.add(user)
    db.flush()

    profile = SysUserProfile(
        user_id=user.user_id,
        create_time=datetime.now(),
    )
    db.add(profile)

    db.commit()
    db.refresh(user)
    return user


def update_user(db: Session, user_id: int, user_in: UserUpdate) -> SysUser:
    user = get_user_or_404(db, user_id)
    update_data = user_in.model_dump(exclude_unset=True)

    if "role_id" in update_data:
        ensure_role_exists(db, update_data["role_id"])

    if "username" in update_data:
        ensure_username_available(db, update_data["username"], exclude_user_id=user_id)

    if "password" in update_data and update_data["password"]:
        update_data["password"] = get_password_hash(update_data["password"])

    for field, value in update_data.items():
        setattr(user, field, value)

    db.add(user)
    db.commit()
    db.refresh(user)
    return user


def update_user_status(db: Session, user_id: int, status: int) -> SysUser:
    user = get_user_or_404(db, user_id)
    user.status = status
    db.add(user)
    db.commit()
    db.refresh(user)
    return user


def get_or_create_profile(db: Session, user_id: int) -> SysUserProfile:
    get_user_or_404(db, user_id)
    profile = db.scalar(select(SysUserProfile).where(SysUserProfile.user_id == user_id))
    if profile:
        return profile

    profile = SysUserProfile(user_id=user_id, create_time=datetime.now())
    db.add(profile)
    db.commit()
    db.refresh(profile)
    return profile


def update_user_profile(
    db: Session,
    user_id: int,
    profile_in: UserProfileUpdate,
) -> SysUserProfile:
    profile = get_or_create_profile(db, user_id)
    update_data = profile_in.model_dump(exclude_unset=True)

    for field, value in update_data.items():
        setattr(profile, field, value)

    db.add(profile)
    db.commit()
    db.refresh(profile)
    return profile


def list_user_stores(db: Session, user_id: int) -> list[UserStore]:
    get_user_or_404(db, user_id)
    stmt = select(UserStore).where(UserStore.user_id == user_id).order_by(UserStore.id.asc())
    return list(db.scalars(stmt).all())


def bind_user_stores(
    db: Session,
    user_id: int,
    bind_in: UserStoreBind,
) -> list[UserStore]:
    user = get_user_or_404(db, user_id)
    role_id = bind_in.role_id or user.role_id
    ensure_role_exists(db, role_id)

    unique_store_ids = list(dict.fromkeys(bind_in.store_ids))
    for store_id in unique_store_ids:
        store = db.get(Store, store_id)
        if not store:
            raise BusinessException(message=f"store not found: {store_id}", code=404)

    db.execute(delete(UserStore).where(UserStore.user_id == user_id))

    for store_id in unique_store_ids:
        db.add(
            UserStore(
                user_id=user_id,
                role_id=role_id,
                store_id=store_id,
            )
        )

    db.commit()
    return list_user_stores(db, user_id)
```

### 注意事项

1. 创建用户时自动创建一条用户档案。
2. 修改用户时如果传了密码，会重新哈希。
3. 用户门店绑定会先清空旧绑定，再写入新绑定。
4. `dict.fromkeys` 用于保持顺序并去重。

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from app.core.exceptions import BusinessException
from app.models.store import Store, StoreProvince, UserStore
from app.schemas.store import ProvinceCreate, ProvinceUpdate, StoreCreate, StoreUpdate
from app.utils.pagination import paginate


def list_provinces(db: Session) -> list[StoreProvince]:
    stmt = select(StoreProvince).order_by(StoreProvince.province_id.asc())
    return list(db.scalars(stmt).all())


def get_province_or_404(db: Session, province_id: int) -> StoreProvince:
    province = db.get(StoreProvince, province_id)
    if not province:
        raise BusinessException(message="province not found", code=404)
    return province


def create_province(db: Session, province_in: ProvinceCreate) -> StoreProvince:
    province = StoreProvince(**province_in.model_dump())
    db.add(province)
    db.commit()
    db.refresh(province)
    return province


def update_province(
    db: Session,
    province_id: int,
    province_in: ProvinceUpdate,
) -> StoreProvince:
    province = get_province_or_404(db, province_id)
    update_data = province_in.model_dump(exclude_unset=True)

    for field, value in update_data.items():
        setattr(province, field, value)

    db.add(province)
    db.commit()
    db.refresh(province)
    return province


def delete_province(db: Session, province_id: int) -> None:
    province = get_province_or_404(db, province_id)
    store_count = db.scalar(
        select(func.count()).select_from(Store).where(Store.province_id == province_id)
    )
    if store_count:
        raise BusinessException(message="province has stores, cannot delete", code=400)

    db.delete(province)
    db.commit()


def list_stores(
    db: Session,
    page: int = 1,
    page_size: int = 10,
    keyword: str | None = None,
    province_id: int | None = None,
    store_type: int | None = None,
    is_operating: int | None = None,
):
    stmt = select(Store).order_by(Store.store_id.desc())

    if keyword:
        like_keyword = f"%{keyword}%"
        stmt = stmt.where(Store.store_name.like(like_keyword))
    if province_id is not None:
        stmt = stmt.where(Store.province_id == province_id)
    if store_type is not None:
        stmt = stmt.where(Store.store_type == store_type)
    if is_operating is not None:
        stmt = stmt.where(Store.is_operating == is_operating)

    return paginate(db, stmt, page=page, page_size=page_size)


def get_store_or_404(db: Session, store_id: int) -> Store:
    store = db.get(Store, store_id)
    if not store:
        raise BusinessException(message="store not found", code=404)
    return store


def create_store(db: Session, store_in: StoreCreate) -> Store:
    province = get_province_or_404(db, store_in.province_id)
    store_data = store_in.model_dump()
    if not store_data.get("province_name"):
        store_data["province_name"] = province.province_name

    store = Store(**store_data)
    db.add(store)
    db.commit()
    db.refresh(store)
    return store


def update_store(db: Session, store_id: int, store_in: StoreUpdate) -> Store:
    store = get_store_or_404(db, store_id)
    update_data = store_in.model_dump(exclude_unset=True)

    if "province_id" in update_data:
        province = get_province_or_404(db, update_data["province_id"])
        if not update_data.get("province_name"):
            update_data["province_name"] = province.province_name

    for field, value in update_data.items():
        setattr(store, field, value)

    db.add(store)
    db.commit()
    db.refresh(store)
    return store


def delete_store(db: Session, store_id: int) -> None:
    store = get_store_or_404(db, store_id)
    bind_count = db.scalar(
        select(func.count()).select_from(UserStore).where(UserStore.store_id == store_id)
    )
    if bind_count:
        raise BusinessException(message="store has user bindings, cannot delete", code=400)

    db.delete(store)
    db.commit()

### 2. 创建门店提示 province not found

### 3. 删除省份提示 province has stores, cannot delete

该省份下已经有门店。

修复：

先删除或迁移该省份下的门店。

### 4. 删除门店提示 store has user bindings, cannot delete

```text

```

清空对应用户的门店绑定。

### 5. 创建用户提示 username already exists

原因：

用户名已存在。

修复：

换一个用户名，或者修改已有用户。

### 问题 1：Swagger 调用用户或门店接口返回 401 Not authenticated

#### 报错现象

在 Swagger 中调用用户、门店、省份等接口时返回：

```json
{
  "code": 401,
  "message": "Not authenticated",
  "data": null
}
```

页面上显示：

```text
401 Error: Unauthorized
```

#### 原因分析

第四阶段的用户和门店接口都接入了 RBAC 权限校验，例如：

```python
current_user: SysUser = Depends(require_permission("user:list"))
```

或者：

```python
current_user: SysUser = Depends(require_permission("store:list"))
```

这些接口必须携带 Token 才能访问。

如果 Swagger 请求中没有请求头：

```text
Authorization: Bearer 你的token
```

后端就会返回：

```text
401 Not authenticated
```

#### 修复办法 1：使用 Swagger Authorize 登录

打开：

```text
http://127.0.0.1:8000/docs
```

点击右上角：

```text
Authorize
```

填写：

```text
username: admin
password: 123456
client_id: 留空
client_secret: 留空
```

然后点击：

```text
Authorize
```

如果授权成功，弹窗里会显示已授权状态。

关闭弹窗后，再调用：

```text
GET /api/users
GET /api/provinces
GET /api/stores
```

此时 Swagger 会自动携带 Token。

#### 修复办法 2：确认 Swagger 使用的是 /api/auth/token

如果 Authorize 弹窗登录失败，需要检查：

相对路径：

```text
backend/app/api/deps.py
```

确认是：

```python
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/auth/token")
```

不要写成：

```python
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/auth/login")
```

原因：

`/api/auth/token` 是给 Swagger OAuth2 Password Flow 使用的表单登录接口。

`/api/auth/login` 是给前端使用的 JSON 登录接口。

#### 修复办法 3：确认 /api/auth/token 接口存在

相对路径：

```text
backend/app/api/v1/auth.py
```

确认存在：

```python
@router.post("/token")
def swagger_login(
    form_data: OAuth2PasswordRequestForm = Depends(),
    db: Session = Depends(get_db),
) -> dict:
    user = authenticate_user(db, form_data.username, form_data.password)
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="username or password error",
        )

    return {
        "access_token": create_access_token(subject=str(user.user_id)),
        "token_type": "bearer",
    }
```

#### 修复办法 4：如果手动调用接口

如果不用 Swagger，而是用 Apifox、Postman、curl 调用，必须先调用：

```text
POST /api/auth/login
```

拿到：

```text
access_token
```

然后在后续请求头中加入：

```text
Authorization: Bearer access_token的值
```

例如：

```bash
curl -H "Authorization: Bearer 你的token" http://127.0.0.1:8000/api/users
```

#### 检查顺序

1. 先确认管理员可以登录 `/api/auth/login`。
2. 再确认 Swagger Authorize 使用的是 `/api/auth/token`。
3. 再确认 Authorize 弹窗中填写的是 `admin` 和 `123456`。
4. 最后再调用受保护接口。

### 问题 2：创建用户返回 role not found

#### 报错现象

调用：

```text
POST /api/users
```

请求体类似：

```json
{
  "role_id": 4,
  "username": "member001",
  "password": "123456",
  "real_name": "测试会员",
  "phone": "13800138000",
  "status": 1
}
```

返回：

```json
{
  "code": 404,
  "message": "role not found",
  "data": null
}
```

#### 原因分析

这个错误说明后端已经通过了 Token 和权限校验，真正失败的位置是创建用户时的角色校验：

```python
ensure_role_exists(db, user_in.role_id)
```

也就是说：

```text
sys_role 表里不存在 id = 4 的角色
```

注意：不要假设 `member` 角色的 ID 一定是 4。

如果数据库之前初始化过、清理过、重复建过、删除过数据，MySQL 的自增 ID 可能不是 1、2、3、4。

#### 修复办法 1：先查询实际角色 ID

在 Swagger 中先调用：

```text
GET /api/system/roles
```

查看当前数据库真实角色列表。

返回示例：

```json
{
  "code": 200,
  "message": "success",
  "data": [
    {
      "id": 1,
      "role_code": "admin",
      "role_name": "超级管理员"
    },
    {
      "id": 2,
      "role_code": "store_admin",
      "role_name": "门店管理员"
    },
    {
      "id": 3,
      "role_code": "coach",
      "role_name": "教练"
    },
    {
      "id": 4,
      "role_code": "member",
      "role_name": "普通会员"
    }
  ]
}
```

如果你的返回里 `member` 不是 `id=4`，创建用户时要使用实际 ID。

例如如果返回：

```json
{
  "id": 7,
  "role_code": "member",
  "role_name": "普通会员"
}
```

那创建用户时应传：

```json
{
  "role_id": 7,
  "username": "member001",
  "password": "123456",
  "real_name": "测试会员",
  "phone": "13800138000",
  "status": 1
}
```

#### 修复办法 2：如果 member 角色不存在，先创建角色

如果 `GET /api/system/roles` 中没有：

```text
role_code = member
```

则先调用：

```text
POST /api/system/roles
```

请求体：

```json
{
  "role_code": "member",
  "role_name": "普通会员"
}
```

创建成功后，使用返回的 `id` 创建用户。

#### 修复办法 3：如果角色初始化数据不完整，检查自动种子配置

检查：

相对路径：

```text
backend/.env
```

确认存在：

```env
AUTO_SEED_DATA=true
```

检查：

相对路径：

```text
backend/app/db/init_db.py
```

确认启动时会执行：

```python
seed_data_if_needed()
```

检查：

相对路径：

```text
backend/app/db/seed.py
```

确认 `DEFAULT_ROLES` 包含：

```python
("member", "普通会员")
```

然后重启后端：

```bash
uvicorn app.main:app --reload --host 0.0.0.0 --port 8000
```

#### 额外说明：为什么 HTTP 状态码显示 200，但 body 里 code 是 404

当前 `BusinessException` 处理器会返回 HTTP 200，但响应体里写业务错误码：

```json
{
  "code": 404,
  "message": "role not found",
  "data": null
}
```

这代表“请求到达后端并被正常处理，但业务校验失败”。

如果你希望 HTTP 状态码也返回 404，可以后续调整：

相对路径：

```text
backend/app/core/exceptions.py
```

把业务异常的：

```python
status_code=status.HTTP_200_OK
```

改为：

```python
status_code=exc.code
```

不过这属于接口风格选择，不影响当前问题修复。

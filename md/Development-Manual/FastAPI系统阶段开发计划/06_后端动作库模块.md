# 06_后端动作库模块

## 一、阶段目标

本阶段目标是在课程模块已经完成的基础上，实现动作分类、动作库条目、用户动作收藏等后端接口。

完成本阶段后，后端应具备以下能力：

1. 管理员可以维护动作分类。
2. 管理员可以维护动作库条目。
3. 管理员可以按分类、难度、适用门店类型、关键字筛选动作。
4. 登录用户可以收藏动作。
5. 登录用户可以取消收藏动作。
6. 登录用户可以查看自己的动作收藏列表。
7. 动作管理接口接入 `action:*` 权限。
8. 用户动作收藏接口只要求登录，不额外要求后台权限。

## 二、本阶段开发顺序

```text
1. 创建动作库 Schema
2. 创建动作库 Service
3. 创建动作库 API 路由
4. 注册动作库路由
5. 修改路由注册测试
6. 启动服务并在 Swagger 验证接口
```

```python
from datetime import datetime

from pydantic import BaseModel, Field

from app.schemas.common import ORMBaseModel


class ActionCategoryCreate(BaseModel):
    category_name: str = Field(min_length=1, max_length=64, description="分类名称")
    category_image_url: str | None = Field(
        default=None,
        max_length=255,
        description="分类图标/图片",

class ActionCategoryUpdate(BaseModel):
    category_name: str | None = Field(default=None, min_length=1, max_length=64)
    category_image_url: str | None = Field(default=None, max_length=255)


class ActionCategoryRead(ORMBaseModel):
    category_id: int
    category_name: str
    category_image_url: str | None = None


class ActionCreate(BaseModel):
    action_name: str = Field(min_length=1, max_length=64, description="动作名称")
    category_id: int = Field(description="动作分类ID")
    action_difficulty: int | None = Field(
        default=None,
        ge=1,
        le=5,
        description="动作难度：1-5",
    )
    action_image_url: str | None = Field(
        default=None,
        max_length=255,
        description="动作演示图/视频封面",
    )
    action_steps: str | None = Field(default=None, description="动作步骤文案")
    attention_points: str | None = Field(default=None, description="注意事项")
    applicable_equipment: str | None = Field(
        default=None,
        max_length=255,
        description="适用器械",
    )
    applicable_store_type: int | None = Field(
        default=None,
        ge=1,
        le=2,
        description="适用门店类型：1铁馆，2商业私教馆",
    )


class ActionUpdate(BaseModel):
    action_name: str | None = Field(default=None, min_length=1, max_length=64)
    category_id: int | None = None
    action_difficulty: int | None = Field(default=None, ge=1, le=5)
    action_image_url: str | None = Field(default=None, max_length=255)
    action_steps: str | None = None
    attention_points: str | None = None
    applicable_equipment: str | None = Field(default=None, max_length=255)
    applicable_store_type: int | None = Field(default=None, ge=1, le=2)


class ActionRead(ORMBaseModel):
    action_id: int
    action_name: str
    category_id: int
    action_difficulty: int | None = None
    action_image_url: str | None = None
    action_steps: str | None = None
    attention_points: str | None = None
    applicable_equipment: str | None = None
    applicable_store_type: int | None = None


class ActionFavoriteRead(ORMBaseModel):
    favorite_id: int
    user_id: int
    action_id: int
    created_at: datetime | None = None
```

### 注意事项

1. `action_difficulty` 建议限定为 1 到 5。
2. `applicable_store_type` 对应门店类型：`1` 表示铁馆，`2` 表示商业私教馆。
3. `action_steps` 和 `attention_points` 暂时用文本保存，后续前端可以按换行展示。

## 七、步骤 2：创建动作库 Service

### 相对路径

```text
backend/app/services/action.py
```

### 完整代码

```python
from sqlalchemy import func, select
from sqlalchemy.orm import Session

from app.core.exceptions import BusinessException
from app.models.action import Action, ActionCategory, UserActionFavorite
from app.schemas.action import (
    ActionCategoryCreate,
    ActionCategoryUpdate,
    ActionCreate,
    ActionUpdate,
)
from app.utils.pagination import paginate


def list_action_categories(db: Session) -> list[ActionCategory]:
    stmt = select(ActionCategory).order_by(ActionCategory.category_id.asc())
    return list(db.scalars(stmt).all())


def get_action_category_or_404(db: Session, category_id: int) -> ActionCategory:
    category = db.get(ActionCategory, category_id)
    if not category:
        raise BusinessException(message="action category not found", code=404)
    return category


def create_action_category(
    db: Session,
    category_in: ActionCategoryCreate,
) -> ActionCategory:
    category = ActionCategory(**category_in.model_dump())
    db.add(category)
    db.commit()
    db.refresh(category)
    return category


def update_action_category(
    db: Session,
    category_id: int,
    category_in: ActionCategoryUpdate,
) -> ActionCategory:
    category = get_action_category_or_404(db, category_id)
    update_data = category_in.model_dump(exclude_unset=True)

    for field, value in update_data.items():
        setattr(category, field, value)

    db.add(category)
    db.commit()
    db.refresh(category)
    return category


def delete_action_category(db: Session, category_id: int) -> None:
    category = get_action_category_or_404(db, category_id)
    action_count = db.scalar(
        select(func.count()).select_from(Action).where(Action.category_id == category_id)
    )
    if action_count:
        raise BusinessException(message="category has actions, cannot delete", code=400)

    db.delete(category)
    db.commit()


def list_actions(
    db: Session,
    page: int = 1,
    page_size: int = 10,
    keyword: str | None = None,
    category_id: int | None = None,
    action_difficulty: int | None = None,
    applicable_store_type: int | None = None,
):
    stmt = select(Action).order_by(Action.action_id.desc())

    if keyword:
        stmt = stmt.where(Action.action_name.like(f"%{keyword}%"))
    if category_id is not None:
        stmt = stmt.where(Action.category_id == category_id)
    if action_difficulty is not None:
        stmt = stmt.where(Action.action_difficulty == action_difficulty)
    if applicable_store_type is not None:
        stmt = stmt.where(Action.applicable_store_type == applicable_store_type)

    return paginate(db, stmt, page=page, page_size=page_size)


def get_action_or_404(db: Session, action_id: int) -> Action:
    action = db.get(Action, action_id)
    if not action:
        raise BusinessException(message="action not found", code=404)
    return action


def create_action(db: Session, action_in: ActionCreate) -> Action:
    get_action_category_or_404(db, action_in.category_id)

    action = Action(**action_in.model_dump())
    db.add(action)
    db.commit()
    db.refresh(action)
    return action


def update_action(
    db: Session,
    action_id: int,
    action_in: ActionUpdate,
) -> Action:
    action = get_action_or_404(db, action_id)
    update_data = action_in.model_dump(exclude_unset=True)

    if "category_id" in update_data:
        get_action_category_or_404(db, update_data["category_id"])

    for field, value in update_data.items():
        setattr(action, field, value)

    db.add(action)
    db.commit()
    db.refresh(action)
    return action


def delete_action(db: Session, action_id: int) -> None:
    action = get_action_or_404(db, action_id)
    favorite_count = db.scalar(
        select(func.count()).select_from(UserActionFavorite).where(
            UserActionFavorite.action_id == action_id
        )
    )
    if favorite_count:
        raise BusinessException(message="action has favorites, cannot delete", code=400)

    db.delete(action)
    db.commit()


def favorite_action(
    db: Session,
    user_id: int,
    action_id: int,
) -> UserActionFavorite:
    get_action_or_404(db, action_id)

    favorite = db.scalar(
        select(UserActionFavorite).where(
            UserActionFavorite.user_id == user_id,
            UserActionFavorite.action_id == action_id,
        )
    )
    if favorite:
        return favorite

    favorite = UserActionFavorite(user_id=user_id, action_id=action_id)
    db.add(favorite)
    db.commit()
    db.refresh(favorite)
    return favorite


def unfavorite_action(
    db: Session,
    user_id: int,
    action_id: int,
) -> None:
    favorite = db.scalar(
        select(UserActionFavorite).where(
            UserActionFavorite.user_id == user_id,
            UserActionFavorite.action_id == action_id,
        )
    )
    if not favorite:
        raise BusinessException(message="action favorite not found", code=404)

    db.delete(favorite)
    db.commit()


def list_my_action_favorites(db: Session, user_id: int) -> list[UserActionFavorite]:
    stmt = (
        select(UserActionFavorite)
        .where(UserActionFavorite.user_id == user_id)
        .order_by(UserActionFavorite.favorite_id.desc())
    )
    return list(db.scalars(stmt).all())
```

### 注意事项

1. 删除动作分类前检查该分类下是否已有动作。
2. 删除动作前检查该动作是否已被用户收藏。
3. 收藏接口重复调用不会重复插入，直接返回已有收藏。
4. 取消收藏时如果收藏记录不存在，返回业务错误。

```python
from sqlalchemy.orm import Session

from app.api.deps import get_current_user, require_permission
from app.db.session import get_db
from app.models.action import Action, ActionCategory, UserActionFavorite
from app.models.user import SysUser

from app.utils.response import success


router = APIRouter(tags=["动作库管理"])


def category_to_dict(item: ActionCategory) -> dict:
    return {
        "category_id": item.category_id,
        "category_name": item.category_name,
        "category_image_url": item.category_image_url,
    }


def action_to_dict(item: Action) -> dict:
    return {
        "action_id": item.action_id,
        "action_name": item.action_name,
        "category_id": item.category_id,
        "action_difficulty": item.action_difficulty,
        "action_image_url": item.action_image_url,
        "action_steps": item.action_steps,
        "attention_points": item.attention_points,
        "applicable_equipment": item.applicable_equipment,
        "applicable_store_type": item.applicable_store_type,
    }


def favorite_to_dict(item: UserActionFavorite) -> dict:
    return {
        "favorite_id": item.favorite_id,
        "user_id": item.user_id,
        "action_id": item.action_id,
        "created_at": item.created_at,
    }


@router.get("/action-categories")
def get_action_categories(
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("action:list")),
) -> dict:
    items = list_action_categories(db)
    return success(data=[category_to_dict(item) for item in items])


@router.post("/action-categories")
def add_action_category(
    payload: ActionCategoryCreate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("action:create")),
) -> dict:
    category = create_action_category(db, payload)
    return success(data=category_to_dict(category))


@router.put("/action-categories/{category_id}")
def edit_action_category(
    category_id: int,
    payload: ActionCategoryUpdate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("action:update")),
) -> dict:
    category = update_action_category(db, category_id, payload)
    return success(data=category_to_dict(category))


@router.delete("/action-categories/{category_id}")
def remove_action_category(
    category_id: int,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("action:delete")),
) -> dict:
    delete_action_category(db, category_id)
    return success(message="deleted")


@router.get("/actions")
def get_actions(
    page: int = Query(default=1, ge=1),
    page_size: int = Query(default=10, ge=1, le=100),
    keyword: str | None = Query(default=None),
    category_id: int | None = Query(default=None),
    action_difficulty: int | None = Query(default=None),
    applicable_store_type: int | None = Query(default=None),
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("action:list")),
) -> dict:
    result = list_actions(
        db,
        page=page,
        page_size=page_size,
        keyword=keyword,
        category_id=category_id,
        action_difficulty=action_difficulty,
        applicable_store_type=applicable_store_type,
    )
    return success(
        data={
            "items": [action_to_dict(item) for item in result.items],
            "total": result.total,
            "page": result.page,
            "page_size": result.page_size,
            "pages": result.pages,
        }
    )


@router.post("/actions")
def add_action(
    payload: ActionCreate,
    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("action:create")),
) -> dict:

    return success(data=action_to_dict(action))


@router.get("/actions/{action_id}")
def get_action_detail(

    db: Session = Depends(get_db),
    current_user: SysUser = Depends(require_permission("action:list")),
) -> dict:

    return success(data=action_to_dict(action))


@router.put("/actions/{action_id}")
def edit_action(

    db: Session = Depends(get_db),
```

## 十四、常见问题

### 1. 创建动作提示 action category not found

原因：

传入的 `category_id` 不存在。

修复：

先调用：

```text
POST /api/action-categories
```

创建动作分类，再使用返回的 `category_id` 创建动作。

### 2. 删除动作分类提示 category has actions, cannot delete

原因：

该分类下已经有动作。

修复：

先删除或迁移该分类下的动作。

### 3. 删除动作提示 action has favorites, cannot delete

原因：

动作已经被用户收藏。

修复：

先取消相关收藏，或改成后台逻辑删除动作。

### 4. 收藏动作提示 action not found

原因：

路径中的 `action_id` 不存在。

修复：

先调用：

```text
GET /api/actions
```

找到真实存在的 `action_id`。

### 5. 后台动作接口返回 403 permission denied

原因：

当前用户角色没有 `action:list`、`action:create`、`action:update` 或 `action:delete` 权限。

修复：

使用管理员账号测试，或给当前角色分配动作库权限。

### 6. Swagger 中创建动作返回 422

原因：

请求体字段校验不通过，常见情况包括：

1. `action_difficulty` 不在 1 到 5 范围内。
2. `applicable_store_type` 不在 1 到 2 范围内。
3. `action_name` 为空。
4. 图片 URL 或器械字段超过长度限制。

修复：

按照 Schema 约束修改请求体后重新提交。
